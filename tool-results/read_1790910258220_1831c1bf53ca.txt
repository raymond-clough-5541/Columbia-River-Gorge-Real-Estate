    57→interface SavedScenario {
    58→  id: string;
    59→  name: string;
    60→  selected: string[];
    61→  pv: number;
    62→  cagr: number;
    63→  horizon: number;
    64→  showScenario: boolean;
    65→  savedAt: string;
    66→}
    67→
    68→const STORAGE_KEY = "crgnsa-saved-scenarios";
    69→const CHANGE_EVENT = "crgnsa:saved-scenarios-changed";
    70→const EMPTY: SavedScenario[] = [];
    71→
    72→let cachedRaw: string | null = null;
    73→let cachedList: SavedScenario[] = EMPTY;
    74→
    75→function readSaved(): SavedScenario[] {
    76→  let raw: string | null = null;
    77→  try {
    78→    raw = window.localStorage.getItem(STORAGE_KEY);
    79→  } catch {
    80→    raw = null;
    81→  }
    82→  if (raw === cachedRaw) return cachedList;
    83→  cachedRaw = raw;
    84→  try {
    85→    const parsed = raw ? (JSON.parse(raw) as SavedScenario[]) : [];
    86→    cachedList = Array.isArray(parsed) ? parsed : EMPTY;
    87→  } catch {
    88→    cachedList = EMPTY;
    89→  }
    90→  return cachedList;
    91→}
    92→
    93→function writeSaved(list: SavedScenario[]): void {
    94→  try {
    95→    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
    96→    window.dispatchEvent(new Event(CHANGE_EVENT));
    97→  } catch {
    98→    /* storage unavailable — ignore */
    99→  }
   100→}
   101→
   102→function useSavedScenarios(): SavedScenario[] {
   103→  return useSyncExternalStore(
   104→    (cb) => {
   105→      window.addEventListener(CHANGE_EVENT, cb);
   106→      window.addEventListener("storage", cb);
   107→      return () => {
   108→        window.removeEventListener(CHANGE_EVENT, cb);
   109→        window.removeEventListener("storage", cb);
   110→      };
   111→    },
   112→    readSaved,
   113→    () => EMPTY
   114→  );
   115→}

   117→interface TooltipEntry {
   118→  name?: string;
   119→  value?: number;
   120→  dataKey?: string | number;
   121→  color?: string;
   122→}
   123→
   124→function ChartTooltip({
   125→  active,
   126→  payload,
   127→  label,
   128→  depletionBySlug,
   129→}: {
   130→  active?: boolean;
   131→  payload?: TooltipEntry[];
   132→  label?: string | number;
   133→  depletionBySlug: Record<string, number>;
   134→}) {
   135→  if (!active || !payload || payload.length === 0) return null;
   136→  const year = Number(label);
   137→  return (
   138→    <div className="rounded-lg border bg-popover p-3.5 text-popover-foreground shadow-xl">
   139→      <p className="text-[12px] font-semibold uppercase tracking-wider text-muted-foreground">
   140→        {year}
   141→      </p>
   142→      <div className="mt-2 space-y-1.5">
   143→        {payload.map((entry) => {
   144→          const value = entry.value ?? 0;
   145→          const isScenario = entry.dataKey === "scenario";
   146→          const depletion = isScenario
   147→            ? null
   148→            : depletionBySlug[String(entry.dataKey)];
   149→          const depleted = depletion !== null && depletion !== undefined && year >= depletion;
   150→          return (
   151→            <div key={String(entry.dataKey)} className="flex items-center gap-2.5 text-[13px]">
   152→              <span
   153→                className="h-2.5 w-2.5 shrink-0 rounded-full"
   154→                style={{ backgroundColor: entry.color }}
   155→                aria-hidden
   156→              />
   157→              <span className="font-medium">{entry.name}</span>
   158→              <span className="ml-auto pl-4 font-semibold tabular-nums">
   159→                {fmtCurrency(value, { compact: true })}
   160→              </span>
   161→              {depleted ? (
   162→                <Flame className="h-3.5 w-3.5 shrink-0 text-rose-500" aria-label="Raw land depleted" />
   163→              ) : null}
   164→            </div>
   165→          );
   166→        })}
   167→      </div>
   168→      <p className="mt-2 border-t pt-1.5 text-[11px] text-muted-foreground">
   169→        🜂 marks years after a market's projected raw-land exhaustion
   170→      </p>
   171→    </div>
   172→  );
   173→}

   191→

   195→      const year = START_YEAR + n;

   198→        row[s.slug] = futureValue(s.baselinePrice2026, s.projectedCagr, n);
   199→      }

   201→        row.scenario = futureValue(pv, cagr, n);

   204→    }

   206→  }, [selectedMarkets, horizon, pv, cagr, showScenario]);

   208→  const depletionBySlug = useMemo(() => {
   209→    const map: Record<string, number> = {};

   211→    return map;

   241→  const saveScenario = () => {

   245→        title: "Name your scenario first",
