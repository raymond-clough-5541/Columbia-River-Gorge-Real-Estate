    76→    76→/** Shared id prefix for every hypothetical column ("what-if-1"…"what-if-3").
    77→    77→ *  Live listing ids are Prisma uuids, so a prefix test can't false-positive.

    79→    79→const WHAT_IF_ID = "what-if";
    80→    80→/** Hard cap — three hypotheticals (entry / premium / land play) is where
    81→    81→ *  the sheet stays readable; past that columns stop comparing and start
    82→    82→ *  wallpapering. Matches the matrix's 3-pin convention. */
    83→    83→const WHAT_IF_MAX = 3;
    84→    84→const whatIfId = (slot: number) => `${WHAT_IF_ID}-${slot}`;
    85→    85→const isWhatIf = (id: string) => id.startsWith(`${WHAT_IF_ID}-`);
    86→    86→
    87→    87→/** Smallest unused slot 1..WHAT_IF_MAX, so ids stay canonical AND unique
    88→    88→ *  across add → remove → add churn (["-1", "-3"] refills "-2", never a
    89→    89→ *  dupe). Unreachable overflow slot keeps the helper total. */
    90→    90→const nextWhatIfSlot = (existing: PropertyListing[]): number => {
    91→    91→  const used = new Set(existing.map((w) => w.id));
    92→    92→  for (let slot = 1; slot <= WHAT_IF_MAX; slot += 1) {
    93→    93→    if (!used.has(whatIfId(slot))) return slot;
    94→    94→  }
    95→    95→  return WHAT_IF_MAX + 1;
    96→    96→};

   164→   164→function underwrite(l: PropertyListing): CompareMetrics {
   165→   165→  const isLand = l.propertyType === "Land Parcel" || l.squareFeet === 0;
   166→   166→  const price = Math.max(1, l.price);
   167→   167→  const downPct = isLand ? LAND_DOWN_PCT : SHEET_DOWN_PCT;
   168→   168→  const rate = isLand ? LAND_RATE : SHEET_RATE;
   169→   169→  const down = (price * downPct) / 100;
   170→   170→  const loan = price - down;
   171→   171→  const state = l.submarket?.state ?? "OR";
   172→   172→  const pi = monthlyPayment(loan, rate, SHEET_TERM);
   173→   173→  const tax = (price * PROPERTY_TAX_RATES[state]) / 100 / 12;
   174→   174→  const ins = isLand ? 0 : (price * INSURANCE_RATE) / 12;
   175→   175→  const monthlyCarry = pi + tax + ins;
   176→   176→
   177→   177→  const rent = estimateMarketRent(l); // market-indexed per round 6
   178→   178→  const grossAnnual = (rent.dwelling + rent.agriculture) * 12;
   179→   179→  const noi = grossAnnual * (1 - RENTAL_RESERVE_RATE);
   180→   180→  const cagr = l.submarket?.projectedCagr ?? 5;
   181→   181→  const fv2046 = futureValue(price, cagr, 20);
   182→   182→
   183→   183→  return {
   184→   184→    listing: l,
   185→   185→    isLand,
   186→   186→    loan,
   187→   187→    monthlyCarry,
   188→   188→    rentMonthly: rent.dwelling + rent.agriculture,
   189→   189→    grossYield: isLand ? 0 : (grossAnnual / price) * 100,
   190→   190→    noiYield: isLand ? 0 : (noi / price) * 100,
   191→   191→    cashFlow: noi - monthlyCarry * 12,
   192→   192→    cashOnCash: down > 0 ? (noi - monthlyCarry * 12) / down : 0,
   193→   193→    fv2046,
   194→   194→    multiple: Math.pow(1 + cagr / 100, 20),
   195→   195→    pricePerSqft: !isLand && l.squareFeet > 0 ? price / l.squareFeet : null,
   196→   196→    runway: runwayYears(l.submarket?.depletionYear ?? 2046),
   197→   197→  };
   198→   198→}

   309→   309→/** Pick the winning metric entry for a row (null when informational). */
   310→   310→function winnerOf(
   311→   311→  row: RowDef,
   312→   312→  entries: CompareMetrics[]
   313→   313→): CompareMetrics | null {
   314→   314→  if (!row.better || entries.length < 2) return null;
   315→   315→  const scored = entries.filter((m) => !row.na?.(m));
   316→   316→  if (scored.length < 2) return null;
   317→   317→  const val = (m: CompareMetrics) => {
   318→   318→    switch (row.key) {
   319→   319→      case "price": return m.listing.price;
   320→   320→      case "sqft": return m.pricePerSqft ?? Infinity;
   321→   321→      case "carry": return m.monthlyCarry;
   322→   322→      case "rent": return m.rentMonthly;
   323→   323→      case "gross": return m.grossYield;
   324→   324→      case "noi": return m.noiYield;
   325→   325→      case "cashflow": return m.cashFlow;
   326→   326→      case "coc": return m.cashOnCash;
   327→   327→      case "fv": return m.fv2046;
   328→   328→      case "mult": return m.multiple;
   329→   329→      case "runway": return m.runway;
   330→   330→      default: return 0;
   331→   331→    }
   332→   332→  };
   333→   333→  const sorted = [...scored].sort((a, b) =>
   334→   334→    row.better === "high" ? val(b) - val(a) : val(a) - val(b)
   335→   335→  );
   336→   336→  const best = sorted[0];
   337→   337→  // Ties → both win (highlight stays honest).
   338→   338→  return val(best) === val(sorted[1]) ? null : best;
   339→   339→}

   341→   341→export function CompareSheet({
   342→   342→  listings,
   343→   343→  open,
   344→   344→  onOpenChange,
   345→   345→  onOpenListing,
   346→   346→  submarkets,
   347→   347→}: {
   348→   348→  listings: PropertyListing[];
   349→   349→  open: boolean;
   350→   350→  onOpenChange: (open: boolean) => void;
   351→   351→  onOpenListing?: (listing: PropertyListing) => void;
   352→   352→  /** Real micro-markets the what-if editor can underwrite in. */
   353→   353→  submarkets?: Submarket[];
   354→   354→}) {
   355→   355→  const watchlist = useWatchlist();
   356→   356→  const { toast } = useToast();
   357→   357→
   358→   358→  // Star order = the user's shortlist order.
   359→   359→  const starred = useMemo(
   360→   360→    () =>
   361→   361→      watchlist
   362→   362→        .map((id) => listings.find((l) => l.id === id))
   363→   363→        .filter((l): l is PropertyListing => Boolean(l)),
   364→   364→    [watchlist, listings]
   365→   365→  );
   366→   366→  /* ---- Hypothetical what-if columns (round 8-a) ----
   367→   367→     Up to WHAT_IF_MAX synthetic listings underwrite side by side, in
   368→   368→     creation order. The array lives here — the sheet stays mounted across
   369→   369→     open/close via CompareTrigger — so the hypotheticals survive closing
   370→   370→     and reopening the sheet for the whole session. */
   371→   371→  const [whatIfs, setWhatIfs] = useState<PropertyListing[]>([]);
   372→   372→  const [editorOpen, setEditorOpen] = useState(false);
   373→   373→  const [form, setForm] = useState<WhatIfForm>({
   374→   374→    price: "",
   375→   375→    type: "Single-Family",
   376→   376→    market: "",
   377→   377→    sqft: String(WHAT_IF_DEFAULT_SQFT),
   378→   378→    acreage: String(WHAT_IF_DEFAULT_ACREAGE),
   379→   379→    label: "",
   380→   380→  });
   381→   381→
   382→   382→  const sortedSubmarkets = useMemo(
   383→   383→    () => [...(submarkets ?? [])].sort((a, b) => a.name.localeCompare(b.name)),
   384→   384→    [submarkets]
   385→   385→  );
   386→   386→
   387→   387→  const entries = useMemo(
   388→   388→    () => [...starred.map(underwrite), ...whatIfs.map(underwrite)],
   389→   389→    [starred, whatIfs]
   390→   390→  );
   391→   391→
   392→   392→  const bestIncome = useMemo(() => {
   393→   393→    const improved = entries.filter((m) => !m.isLand);
   394→   394→    if (improved.length === 0) return null;
   395→   395→    return improved.reduce((a, b) => (b.cashFlow > a.cashFlow ? b : a));
   396→   396→  }, [entries]);
   397→   397→  const bestEquity = useMemo(() => {
   398→   398→    if (entries.length === 0) return null;
   399→   399→    return entries.reduce((a, b) => (b.fv2046 > a.fv2046 ? b : a));
   400→   400→  }, [entries]);
   401→   401→
   402→   402→  /* ---- What-if editor plumbing (all toasts fire in event handlers,
   403→   403→     never inside state updaters — the round-6 bug class). The editor is
   404→   404→     ADD-only as of round 8-a: every open resets to starred-derived
   405→   405→     defaults, and the 3-entry limit is gated here so a 4th column can
   406→   406→     never even open the form. ---- */
   407→   407→
   408→   408→  const canWhatIf = starred.length >= 2 && sortedSubmarkets.length > 0;
   409→   409→
   410→   410→  /** Median starred asking price, rounded to the nearest $5k. */
   411→   411→  const defaultWhatIfPrice = useMemo(() => {
   412→   412→    const prices = starred.map((l) => l.price).sort((a, b) => a - b);
   413→   413→    if (prices.length === 0) return 450_000;
   414→   414→    const mid = Math.floor(prices.length / 2);
   415→   415→    const median =
   416→   416→      prices.length % 2 === 1
   417→   417→        ? prices[mid]
   418→   418→        : (prices[mid - 1] + prices[mid]) / 2;
   419→   419→    return Math.round(median / WHAT_IF_PRICE_STEP) * WHAT_IF_PRICE_STEP;
   420→   420→  }, [starred]);
   421→   421→
   422→   422→  const handleEditorOpenChange = (next: boolean) => {
   423→   423→    if (next) {
   424→   424→      // Limit gate at open time (event-handler context, never an updater) —
   425→   425→      // the toast fires once per click and the popover simply stays closed.
   426→   426→      if (whatIfs.length >= WHAT_IF_MAX) {
   427→   427→        toast({
   428→   428→          title: "What-if limit reached",
   429→   429→          description:
   430→   430→            "Three hypotheticals at a time — remove one to add another.",
   431→   431→        });
   432→   432→        return;
   433→   433→      }
   434→   434→      const firstStarredSlug = starred[0]?.submarket?.slug ?? "";
   435→   435→      const defaultSlug = sortedSubmarkets.some(
   436→   436→        (s) => s.slug === firstStarredSlug
   437→   437→      )
   438→   438→        ? firstStarredSlug
   439→   439→        : (sortedSubmarkets[0]?.slug ?? "");
   440→   440→      setForm({
   441→   441→        price: String(defaultWhatIfPrice),
   442→   442→        type: "Single-Family",
   443→   443→        market: defaultSlug,
   444→   444→        sqft: String(WHAT_IF_DEFAULT_SQFT),
   445→   445→        acreage: String(WHAT_IF_DEFAULT_ACREAGE),
   446→   446→        label: "",
   447→   447→      });
   448→   448→    }
   449→   449→    setEditorOpen(next);
   450→   450→  };
   451→   451→
   452→   452→  const submitWhatIf = (e: FormEvent<HTMLFormElement>) => {
   453→   453→    e.preventDefault();
   454→   454→    // Defensive double-gate: the popover can't open at the limit, but a
   455→   455→    // stray Enter (or a future programmatic submit) must not grow a 4th column.
   456→   456→    if (whatIfs.length >= WHAT_IF_MAX) {
