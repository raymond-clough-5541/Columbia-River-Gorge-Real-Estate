    22→/* Map geometry: 0–100 market coordinates → 1000×460 viewBox units. */
    23→const VW = 1000;
    24→const VH = 460;
    25→function pos(x: number, y: number) {
    26→  return { px: 30 + x * 9.4, py: 20 + y * 4.2 };
    27→}
    28→
    29→const TIER_FILL: Record<string, string> = {
    30→  elite: "#10b981",
    31→  strong: "#14b8a6",
    32→  moderate: "#f59e0b",
    33→  baseline: "#a1a1aa",
    34→};
    35→
    36→function tierOf(cagr: number): string {
    37→  if (cagr >= 5.5) return "elite";
    38→  if (cagr >= 5.0) return "strong";
    39→  if (cagr >= 4.7) return "moderate";
    40→  return "baseline";
    41→}
    42→
    43→/** Lens of the map: appreciation tiers vs depletion-runway heat. */
    44→type MapMode = "cagr" | "runway";
    45→
    46→/** Timeline bounds — the 20-year analysis horizon. */
    47→const T0 = 2026;
    48→const T1 = 2046;
    49→/** Wall-clock pace of the play animation: ms per projected year (≈8s sweep). */
    50→const PLAY_MS_PER_YEAR = 400;
    51→/** prefers-reduced-motion pace: discrete one-year steps, no tweening. */
    52→const REDUCED_PLAY_MS_PER_YEAR = 700;
    53→
    54→const MODES: { id: MapMode; label: string; icon: typeof TrendingUp; hint: string }[] = [
    55→  { id: "cagr", label: "Appreciation", icon: TrendingUp, hint: "Dot color = 20-yr CAGR tier" },
    56→  { id: "runway", label: "Land runway", icon: Flame, hint: "Dot color = years of raw land left" },
    57→];
    58→
    59→/** Fill color for a market under the active lens (dark zinc once spent). */
    60→function dotFill(s: Submarket, mode: MapMode, spent: boolean): string {
    61→  if (spent) return "#71717a";
    62→  if (mode === "cagr") return TIER_FILL[tierOf(s.projectedCagr)];
    63→  return RUNWAY_TIER_STYLES[runwayTier(runwayYears(s.depletionYear))].color;
    64→}
    65→

    67→const LABEL_POS: Record<string, { dx: number; dy: number; anchor: "start" | "end" }> = {

    69→  stevenson: { dx: 24, dy: 5, anchor: "start" },

    71→  bingen: { dx: 24, dy: 18, anchor: "start" },

    74→  wishram: { dx: -24, dy: 5, anchor: "end" },
    75→  "cascade-locks": { dx: 24, dy: 5, anchor: "start" },
    76→  mosier: { dx: 24, dy: 5, anchor: "start" },
    77→  "hood-river": { dx: 24, dy: 18, anchor: "start" },
    78→  "the-dalles": { dx: -24, dy: 5, anchor: "end" },
    79→};
    80→

    82→  "M 12 232 C 150 192, 250 272, 390 232 C 520 194, 630 272, 760 232 C 870 198, 940 268, 990 230";
    83→
    84→export function CorridorMap({
    85→  submarkets,
    86→  stats,
    87→  navigate,
    88→}: {
    89→  submarkets: Submarket[];
    90→  stats: CorridorStats;
    91→  navigate: NavigateFn;
    92→}) {
    93→  const [hovered, setHovered] = useState<string | null>(null);
    94→  const [mode, setMode] = useState<MapMode>("cagr");
    95→
    96→  /* ---------------- Depletion-timeline scrubber ---------------- */
    97→  const [timelineYear, setTimelineYear] = useState(T0);
    98→  const [playing, setPlaying] = useState(false);
    99→  // Mirror the year in a ref so the rAF loop can capture its start value
   100→  // without re-arming on every frame (exhaustive-deps stays clean).
   101→  const yearRef = useRef(timelineYear);
   102→  useEffect(() => {
   103→    yearRef.current = timelineYear;
   104→  }, [timelineYear]);
   105→
   106→  // prefers-reduced-motion — matched lazily after mount (matchMedia is
   107→  // browser-only; reading it during render would break SSR/hydration) and
   108→  // kept live via the change event, so a mid-session preference change
   109→  // swaps the play loop in place.
   110→  const [reducedMotion, setReducedMotion] = useState(false);
   111→  useEffect(() => {
   112→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   113→    const sync = () => setReducedMotion(mq.matches);
   114→    sync();
   115→    mq.addEventListener("change", sync);
   116→    return () => mq.removeEventListener("change", sync);
   117→  }, []);
   118→
   119→  useEffect(() => {
   120→    if (!playing) return;
   121→    // Reduced motion: one discrete year per tick — no rAF interpolation,
   122→    // no smooth tweening. Pause/reset/scrub behavior is unchanged.
   123→    if (reducedMotion) {
   124→      let year = yearRef.current;
   125→      const id = window.setInterval(() => {
   126→        year = Math.min(T1, year + 1);
   127→        setTimelineYear(year);
   128→        if (year >= T1) {
   129→          setPlaying(false);
   130→          window.clearInterval(id);
   131→        }
   132→      }, REDUCED_PLAY_MS_PER_YEAR);
   133→      return () => window.clearInterval(id);
   134→    }
   135→    const startedAt = performance.now();
   136→    const from = yearRef.current;
   137→    let raf = 0;
   138→    const tick = (now: number) => {
   139→      const next = Math.min(T1, Math.round(from + (now - startedAt) / PLAY_MS_PER_YEAR));
   140→      setTimelineYear(next);
   141→      if (next >= T1) {
   142→        setPlaying(false);
   143→        return;
   144→      }
   145→      raf = requestAnimationFrame(tick);
   146→    };
   147→    raf = requestAnimationFrame(tick);
   148→    return () => cancelAnimationFrame(raf);
   149→  }, [playing, reducedMotion]);
   150→
   151→  const scrub = (year: number) => {
   152→    setPlaying(false);
   153→    setTimelineYear(year);
   154→  };
   155→  const togglePlay = () => {
   156→    if (playing) {
   157→      setPlaying(false);
   158→      return;
   159→    }
   160→    // At the end of the reel, play restarts from the 2026 baseline.
   161→    if (yearRef.current >= T1) setTimelineYear(T0);
   162→    setPlaying(true);
   163→  };
   164→
   165→  const spentCount = useMemo(
   166→    () => submarkets.filter((s) => timelineYear >= s.depletionYear).length,
   167→    [submarkets, timelineYear]
   168→  );
   169→
   170→  /* Year-synced KPI readouts — every figure is recomputed from the
   171→     submarket data at the scrub year (never hard-coded). */
   172→  const kpis = useMemo(() => {
   173→    // 1 · Median projected corridor price as of the scrub year.
   174→    const projected = submarkets
   175→      .map((s) => futureValue(s.baselinePrice2026, s.projectedCagr, timelineYear - T0))
   176→      .sort((a, b) => a - b);
   177→    const mid = Math.floor(projected.length / 2);
   178→    const medianPrice =
   179→      projected.length === 0
   180→        ? 0
   181→        : projected.length % 2 === 1
   182→          ? projected[mid]
   183→          : (projected[mid - 1] + projected[mid]) / 2;
   184→    const priceDeltaPct =
   185→      stats.regionalMedianPrice > 0
   186→        ? (medianPrice / stats.regionalMedianPrice - 1) * 100
   187→        : 0;
   188→
   189→    // 2 + 3 · Live reserve + next exhaustion, one pass over the corridor.
   190→    let fullReserve = 0;
   191→    let liveReserve = 0;
   192→    let liveMarkets = 0;
   193→    let next: { year: number; name: string; yearsOut: number; tierColor: string } | null =
   194→      null;
   195→    for (const s of submarkets) {
   196→      const acresMid = netBuildableMid(s);
   197→      fullReserve += acresMid;
   198→      if (timelineYear < s.depletionYear) {
   199→        liveReserve += acresMid;
   200→        liveMarkets += 1;
   201→        if (!next || s.depletionYear < next.year) {
   202→          next = {
   203→            year: s.depletionYear,
   204→            name: s.name,
   205→            yearsOut: s.depletionYear - timelineYear,
   206→            tierColor: RUNWAY_TIER_STYLES[runwayTier(s.depletionYear - timelineYear)].color,
   207→          };
   208→        }
   209→      }
   210→    }
   211→    const reservePct = fullReserve > 0 ? (liveReserve / fullReserve) * 100 : 0;
   212→
   213→    return { medianPrice, priceDeltaPct, liveReserve, liveMarkets, reservePct, next };
   214→  }, [submarkets, stats.regionalMedianPrice, timelineYear]);
   215→
   216→  const dots = useMemo(
   217→    () =>
   218→      submarkets.map((s) => {
   219→        const { px, py } = pos(s.mapX, s.mapY);
   220→        const netMid = (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2;
   221→        const r = 9 + netMid * 0.065;
   222→        const tier = tierOf(s.projectedCagr);
   223→        const label = LABEL_POS[s.slug] ?? { dx: 24, dy: 5, anchor: "start" as const };
   224→        return { s, px, py, r, tier, netMid, label };
   225→      }),
   226→    [submarkets]
   227→  );
   228→
   229→  const hoveredDot = dots.find((d) => d.s.slug === hovered) ?? null;
   230→  const orShare = stats.orNetBuildable + stats.waNetBuildable > 0
   231→    ? (stats.orNetBuildable / (stats.orNetBuildable + stats.waNetBuildable)) * 100
   232→    : 50;
   233→
   234→  return (
   235→    <div className="rounded-xl border bg-card p-4 shadow-sm sm:p-6">
   236→      <div className="mb-4 flex flex-wrap items-start justify-between gap-3">
   237→        <div>
   238→          <MicroLabel>Regional Map Summary</MicroLabel>
   239→          <p className="mt-1 text-sm text-muted-foreground">
   240→            All 11 jurisdictions · dot size = net buildable acres ·{" "}
   241→            {mode === "cagr"
   242→              ? "color = 20-yr CAGR tier"
   243→              : "color = raw-land runway remaining"}
   244→            {timelineYear > T0 ? (
   245→              <>
   246→                {" · as of "}
   247→                <span className="font-semibold tabular-nums text-foreground">
   248→                  {timelineYear}
   249→                </span>
   250→              </>
   251→            ) : null}
   252→          </p>
   253→        </div>
   254→        {/* Lens toggle — segmented control */}
   255→        <div
   256→          role="radiogroup"
   257→          aria-label="Map color lens"
   258→          className="flex rounded-lg border bg-background p-0.5"
   259→        >
   260→          {MODES.map((m) => {
   261→            const active = mode === m.id;
   262→            return (
   263→              <button
   264→                key={m.id}
   265→                type="button"
   266→                role="radio"
   267→                aria-checked={active}
   268→                title={m.hint}
   269→                onClick={() => setMode(m.id)}
   270→                className={cn(
   271→                  "inline-flex h-7 items-center gap-1.5 rounded-md px-2.5 text-[12px] font-semibold transition-all active:scale-[0.97]",
   272→                  "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60 focus-visible:ring-offset-1 focus-visible:ring-offset-background",
   273→                  active
   274→                    ? "bg-zinc-900 text-white shadow-sm dark:bg-emerald-500 dark:text-zinc-950"
   275→                    : "text-muted-foreground hover:text-foreground"
   276→                )}
   277→              >
   278→                <m.icon className="h-3.5 w-3.5" aria-hidden />
   279→                {m.label}
   280→              </button>
   281→            );
   282→          })}
   283→        </div>
   284→      </div>
   285→
   286→      <div className="relative overflow-x-auto">
   287→        <div className="relative min-w-[680px]">
   288→          <svg
   289→            viewBox={`0 0 ${VW} ${VH}`}
   290→            className="h-auto w-full"
   291→            role="img"
   292→            aria-label="Map of the Columbia River Gorge corridor showing the 11 analyzed micro-markets on both sides of the river"
   293→          >
   294→            {/* State backdrops */}
   295→            <rect x="0" y="0" width={VW} height="205" rx="10" className="fill-teal-500/[0.04] dark:fill-teal-400/[0.05]" />
   296→            <rect x="0" y="255" width={VW} height={VH - 255} rx="10" className="fill-amber-500/[0.05] dark:fill-amber-400/[0.04]" />
   297→
   298→            <text x="28" y="36" className="fill-zinc-400 dark:fill-zinc-500" fontSize="17" fontWeight="700" letterSpacing="6">
   299→              WASHINGTON
   300→            </text>
   301→            <text x="28" y={VH - 18} className="fill-zinc-400 dark:fill-zinc-500" fontSize="17" fontWeight="700" letterSpacing="6">
   302→              OREGON
   303→            </text>
   304→
   305→            {/* Highways */}
   306→            <path
   307→              d="M 15 300 C 160 285, 300 315, 450 300 C 600 285, 740 315, 985 300"
   308→              className="stroke-zinc-300 dark:stroke-zinc-700"
   309→              strokeWidth="3" strokeDasharray="10 8" fill="none"
   310→            />
   311→            <path
   312→              d="M 15 165 C 160 150, 300 180, 450 165 C 600 150, 740 180, 985 165"
   313→              className="stroke-zinc-300 dark:stroke-zinc-700"
   314→              strokeWidth="3" strokeDasharray="10 8" fill="none"
   315→            />
   316→            <text x="500" y="296" textAnchor="middle" className="fill-zinc-400 dark:fill-zinc-600" fontSize="11" letterSpacing="3">
   317→              I-84
   318→            </text>
   319→            <text x="500" y="161" textAnchor="middle" className="fill-zinc-400 dark:fill-zinc-600" fontSize="11" letterSpacing="3">
   320→              SR-14
   321→            </text>
   322→
   323→            {/* Columbia River */}
   324→            <path
   325→              d={RIVER_PATH}
   326→              className="stroke-sky-200/70 dark:stroke-sky-400/25"
   327→              strokeWidth="46" strokeLinecap="round" fill="none"
   328→            />
   329→            <path
   330→              d={RIVER_PATH}
   331→              className="stroke-sky-300/80 dark:stroke-sky-500/40"
   332→              strokeWidth="4" strokeLinecap="round" fill="none"
   333→            />
   334→            <text
   335→              x="500" y="238" textAnchor="middle" rotate={-3}
   336→              className="fill-sky-500/70 dark:fill-sky-300/60"
   337→              fontSize="13" fontWeight="700" letterSpacing="7"
   338→            >
   339→              COLUMBIA RIVER
   340→            </text>
   341→
   342→            {/* Market dots */}
   343→            {dots.map(({ s, px, py, r, label }) => {
   344→              const active = hovered === s.slug;
   345→              const spent = timelineYear >= s.depletionYear;
   346→              const fill = dotFill(s, mode, spent);
   347→              const runway = runwayYears(s.depletionYear);
   348→              return (
   349→                <g
   350→                  key={s.slug}
   351→                  className="cursor-pointer transition-opacity"
   352→                  onClick={() => navigate({ view: "submarket", slug: s.slug })}
   353→                  onMouseEnter={() => setHovered(s.slug)}
   354→                  onMouseLeave={() => setHovered(null)}
   355→                  role="button"
   356→                  aria-label={`Open ${s.name}, ${s.state} micro-market profile`}
   357→                >
   358→                  <circle cx={px} cy={py} r={r + 15} fill="transparent" />
   359→                  {active ? (
   360→                    <circle cx={px} cy={py} r={r + 9} fill={fill} opacity="0.22" />
   361→                  ) : null}
   362→                  {spent ? (
   363→                    <circle
   364→                      cx={px} cy={py} r={r + 5.5}
   365→                      fill="none"
   366→                      stroke="#71717a"
   367→                      strokeWidth="1.5"
   368→                      strokeDasharray="3 4"
   369→                      opacity={active ? 0.8 : 0.55}
   370→                    />
   371→                  ) : null}
   372→                  <circle
   373→                    cx={px} cy={py} r={r}
   374→                    fill={fill}
   375→                    opacity={spent ? (active ? 0.7 : 0.42) : active ? 1 : 0.88}
   376→                    className="stroke-white dark:stroke-zinc-950"
   377→                    strokeWidth="2.5"
   378→                  />
   379→                  <text
   380→                    x={px + label.dx} y={py + label.dy}
   381→                    textAnchor={label.anchor}
   382→                    fontSize="16.5"
   383→                    fontWeight={active ? 700 : 600}
   384→                    className={cn(
   385→                      active
   386→                        ? "fill-zinc-900 dark:fill-zinc-50"
   387→                        : "fill-zinc-500 dark:fill-zinc-300",
   388→                      spent && !active && "fill-zinc-400/80 dark:fill-zinc-500"
   389→                    )}
   390→                  >
   391→                    {s.slug === "north-bonneville" ? "N. Bonneville" : s.name}
   392→                  </text>
   393→                  <text
   394→                    x={px + label.dx} y={py + label.dy + 16}
   395→                    textAnchor={label.anchor}
   396→                    fontSize="12.5"
   397→                    className={cn(
   398→                      "tabular-nums",
   399→                      spent
   400→                        ? "fill-zinc-400/90 dark:fill-zinc-500"
   401→                        : "fill-zinc-400 dark:fill-zinc-500"
   402→                    )}
   403→                  >
   404→                    {spent
   405→                      ? `${s.state} · exhausted ${s.depletionYear}`
   406→                      : mode === "cagr"
   407→                        ? `${s.state} · ${fmtPct(s.projectedCagr)}`
   408→                        : `${s.state} · ${runway} yrs land`}
   409→                  </text>
   410→                </g>
   411→              );
   412→            })}
   413→          </svg>
   414→
   415→          {/* Hover tooltip */}
   416→          {hoveredDot ? (
   417→            <div
   418→              className="pointer-events-none absolute z-10 w-52 -translate-x-1/2 -translate-y-[115%] rounded-lg border bg-popover p-3 text-popover-foreground shadow-xl"
   419→              style={{
   420→                left: `${(hoveredDot.px / VW) * 100}%`,
   421→                top: `${(hoveredDot.py / VH) * 100}%`,
   422→              }}
   423→            >
   424→              <p className="text-sm font-semibold">
   425→                {hoveredDot.s.name},{" "}
   426→                <span className="text-muted-foreground">{hoveredDot.s.state}</span>
   427→              </p>
   428→              <p className="mt-1 text-[12px] text-muted-foreground">
   429→                {hoveredDot.s.jurisdictionType}
   430→              </p>
   431→              <div className="mt-2 space-y-1 text-[12px] tabular-nums">
   432→                <div className="flex justify-between gap-3">
   433→                  <span className="text-muted-foreground">Net buildable</span>
   434→                  <span className="font-medium">
   435→                    {fmtAcres(hoveredDot.netMid)} band
   436→                  </span>
   437→                </div>
   438→                {mode === "cagr" ? (
   439→                  <div className="flex justify-between gap-3">
   440→                    <span className="text-muted-foreground">20-yr CAGR</span>
   441→                    <span className="font-medium text-emerald-600 dark:text-emerald-400">
   442→                      {fmtPct(hoveredDot.s.projectedCagr)}
   443→                    </span>
   444→                  </div>
   445→                ) : (
   446→                  <div className="flex justify-between gap-3">
   447→                    <span className="text-muted-foreground">Land runway</span>
   448→                    <span
   449→                      className="font-medium"
   450→                      style={{
   451→                        color: RUNWAY_TIER_STYLES[
   452→                          runwayTier(runwayYears(hoveredDot.s.depletionYear))
   453→                        ].color,
   454→                      }}
   455→                    >
   456→                      {runwayYears(hoveredDot.s.depletionYear)} yrs ·{" "}
   457→                      {RUNWAY_TIER_STYLES[
   458→                        runwayTier(runwayYears(hoveredDot.s.depletionYear))
   459→                      ].label.toLowerCase()}
   460→                    </span>
   461→                  </div>
   462→                )}
   463→                <div className="flex justify-between gap-3">
   464→                  <span className="text-muted-foreground">Depletion</span>
   465→                  <span className="font-medium">{hoveredDot.s.depletionYear}</span>
   466→                </div>
   467→                {timelineYear > T0 ? (
   468→                  <div className="flex justify-between gap-3">
   469→                    <span className="text-muted-foreground">As of {timelineYear}</span>
   470→                    <span
   471→                      className={cn(
   472→                        "font-medium",
   473→                        timelineYear >= hoveredDot.s.depletionYear
   474→                          ? "text-rose-600 dark:text-rose-400"
   475→                          : "text-emerald-600 dark:text-emerald-400"
   476→                      )}
   477→                    >
   478→                      {timelineYear >= hoveredDot.s.depletionYear
   479→                        ? `exhausted ${timelineYear - hoveredDot.s.depletionYear} yr${
   480→                            timelineYear - hoveredDot.s.depletionYear === 1 ? "" : "s"
   481→                          } prior`
   482→                        : `${hoveredDot.s.depletionYear - timelineYear} yrs of runway left`}
   483→                    </span>
   484→                  </div>
   485→                ) : null}
   486→              </div>
   487→              <p className="mt-2 border-t pt-1.5 text-[11px] text-muted-foreground">
   488→                Click to open profile →
   489→              </p>
   490→            </div>
   491→          ) : null}
   492→        </div>
   493→      </div>
   494→
   495→      {/* Year-synced KPI readouts — recomputed from the corridor data at the scrub year */}
   496→      <div className="mt-4 grid gap-3 sm:grid-cols-3">
   497→        <div className="rounded-lg border bg-muted/30 p-3">
   498→          <div className="flex items-center gap-1.5">
   499→            <Landmark className="h-3.5 w-3.5 shrink-0 text-muted-foreground" aria-hidden />
   500→            <MicroLabel className="min-w-0">Median price · as of {timelineYear}</MicroLabel>
   501→          </div>
   502→          <p className="mt-2 text-[17px] font-semibold tracking-tight tabular-nums">
   503→            {fmtCurrency(kpis.medianPrice, { compact: true })}
   504→          </p>
   505→          <p
   506→            className={cn(
