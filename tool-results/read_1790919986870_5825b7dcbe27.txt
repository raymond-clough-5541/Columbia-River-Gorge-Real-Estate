    76→/** Shared id prefix for every hypothetical column ("what-if-1"…"what-if-3").
    77→ *  Live listing ids are Prisma uuids, so a prefix test can't false-positive.
    78→ *  `isWhatIf` is the single source of truth for verdict/CSV tagging. */
    79→const WHAT_IF_ID = "what-if";
    80→/** Hard cap — three hypotheticals (entry / premium / land play) is where
    81→ *  the sheet stays readable; past that columns stop comparing and start
    82→ *  wallpapering. Matches the matrix's 3-pin convention. */
    83→const WHAT_IF_MAX = 3;
    84→const whatIfId = (slot: number) => `${WHAT_IF_ID}-${slot}`;
    85→const isWhatIf = (id: string) => id.startsWith(`${WHAT_IF_ID}-`);
    86→
    87→/** Smallest unused slot 1..WHAT_IF_MAX, so ids stay canonical AND unique
    88→ *  across add → remove → add churn (["-1", "-3"] refills "-2", never a
    89→ *  dupe). Unreachable overflow slot keeps the helper total. */
    90→const nextWhatIfSlot = (existing: PropertyListing[]): number => {
    91→  const used = new Set(existing.map((w) => w.id));
    92→  for (let slot = 1; slot <= WHAT_IF_MAX; slot += 1) {
    93→    if (!used.has(whatIfId(slot))) return slot;
    94→  }
    95→  return WHAT_IF_MAX + 1;
    96→};

   111→/** Auto label used when the editor's label field is left blank.
   112→ *  Kept compact (compact-currency) so the column header never truncates. */
   113→const autoWhatIfLabel = (price: number) =>
   114→  `What-if ${fmtCurrency(price, { compact: true })}`;
   115→
   116→/** Round 8-a: two what-ifs at the same price point would auto-label
   117→ *  identically and become indistinguishable amber columns — append
   118→ *  " · 2"/" · 3" until the title is free. Custom labels may collide;
   119→ *  that's the user's call, no enforcement. */
   120→const uniqueAutoWhatIfLabel = (
   121→  price: number,
   122→  existing: PropertyListing[]
   123→): string => {
   124→  const base = autoWhatIfLabel(price);
   125→  if (!existing.some((w) => w.title === base)) return base;
   126→  for (let n = 2; n <= WHAT_IF_MAX; n += 1) {
   127→    const candidate = `${base} · ${n}`;
   128→    if (!existing.some((w) => w.title === candidate)) return candidate;
   129→  }
   130→  return base; // unreachable at ≤3 entries — keeps the helper total
   131→};

   133→const clampWhatIfNumber = (raw: string, min: number, max: number): number => {
   134→  const n = Number(raw);
   135→  return Number.isFinite(n) ? Math.min(max, Math.max(min, n)) : min;

   148→interface CompareMetrics {
   149→  listing: PropertyListing;
   150→  isLand: boolean;
   151→  loan: number;
   152→  monthlyCarry: number;
   153→  rentMonthly: number;
   154→  grossYield: number;

   157→  cashOnCash: number;

   164→function underwrite(l: PropertyListing): CompareMetrics {
   165→  const isLand = l.propertyType === "Land Parcel" || l.squareFeet === 0;
   166→  const price = Math.max(1, l.price);
   167→  const downPct = isLand ? LAND_DOWN_PCT : SHEET_DOWN_PCT;
   168→  const rate = isLand ? LAND_RATE : SHEET_RATE;
   169→  const down = (price * downPct) / 100;
   170→  const loan = price - down;
   171→  const state = l.submarket?.state ?? "OR";
   172→  const pi = monthlyPayment(loan, rate, SHEET_TERM);
   173→  const tax = (price * PROPERTY_TAX_RATES[state]) / 100 / 12;
   174→  const ins = isLand ? 0 : (price * INSURANCE_RATE) / 12;
   175→  const monthlyCarry = pi + tax + ins;
   176→
   177→  const rent = estimateMarketRent(l); // market-indexed per round 6
   178→  const grossAnnual = (rent.dwelling + rent.agriculture) * 12;
   179→  const noi = grossAnnual * (1 - RENTAL_RESERVE_RATE);
   180→  const cagr = l.submarket?.projectedCagr ?? 5;
   181→  const fv2046 = futureValue(price, cagr, 20);
   182→
   183→  return {
   184→    listing: l,
   185→    isLand,
   186→    loan,
   187→    monthlyCarry,
   188→    rentMonthly: rent.dwelling + rent.agriculture,
   189→    grossYield: isLand ? 0 : (grossAnnual / price) * 100,
   190→    noiYield: isLand ? 0 : (noi / price) * 100,
   191→    cashFlow: noi - monthlyCarry * 12,
   192→    cashOnCash: down > 0 ? (noi - monthlyCarry * 12) / down : 0,
   193→    fv2046,
   194→    multiple: Math.pow(1 + cagr / 100, 20),
   195→    pricePerSqft: !isLand && l.squareFeet > 0 ? price / l.squareFeet : null,
   196→    runway: runwayYears(l.submarket?.depletionYear ?? 2046),
   197→  };
   198→}

   309→/** Pick the winning metric entry for a row (null when informational). */
   310→function winnerOf(
   311→  row: RowDef,
   312→  entries: CompareMetrics[]
   313→): CompareMetrics | null {
   314→  if (!row.better || entries.length < 2) return null;
   315→  const scored = entries.filter((m) => !row.na?.(m));
   316→  if (scored.length < 2) return null;
   317→  const val = (m: CompareMetrics) => {
   318→    switch (row.key) {
   319→      case "price": return m.listing.price;
   320→      case "sqft": return m.pricePerSqft ?? Infinity;
   321→      case "carry": return m.monthlyCarry;
   322→      case "rent": return m.rentMonthly;
   323→      case "gross": return m.grossYield;
   324→      case "noi": return m.noiYield;
   325→      case "cashflow": return m.cashFlow;
   326→      case "coc": return m.cashOnCash;
   327→      case "fv": return m.fv2046;
   328→      case "mult": return m.multiple;
   329→      case "runway": return m.runway;
   330→      default: return 0;
   331→    }
   332→  };
   333→  const sorted = [...scored].sort((a, b) =>
   334→    row.better === "high" ? val(b) - val(a) : val(a) - val(b)
   335→  );
   336→  const best = sorted[0];
   337→  // Ties → both win (highlight stays honest).
   338→  return val(best) === val(sorted[1]) ? null : best;
   339→}

   341→export function CompareSheet({
   342→  listings,
   343→  open,
   344→  onOpenChange,
   345→  onOpenListing,
   346→  submarkets,
   347→}: {
   348→  listings: PropertyListing[];
   349→  open: boolean;
   350→  onOpenChange: (open: boolean) => void;
   351→  onOpenListing?: (listing: PropertyListing) => void;
   352→  /** Real micro-markets the what-if editor can underwrite in. */
   353→  submarkets?: Submarket[];
   354→}) {
   355→  const watchlist = useWatchlist();
   356→  const { toast } = useToast();
   357→
   358→  // Star order = the user's shortlist order.
   359→  const starred = useMemo(
   360→    () =>
   361→      watchlist
   362→        .map((id) => listings.find((l) => l.id === id))
   363→        .filter((l): l is PropertyListing => Boolean(l)),
   364→    [watchlist, listings]
   365→  );
   366→  /* ---- Hypothetical what-if columns (round 8-a) ----
   367→     Up to WHAT_IF_MAX synthetic listings underwrite side by side, in
   368→     creation order. The array lives here — the sheet stays mounted across
   369→     open/close via CompareTrigger — so the hypotheticals survive closing
   370→     and reopening the sheet for the whole session. */
   371→  const [whatIfs, setWhatIfs] = useState<PropertyListing[]>([]);
   372→  const [editorOpen, setEditorOpen] = useState(false);
   373→  const [form, setForm] = useState<WhatIfForm>({
   374→    price: "",
   375→    type: "Single-Family",
   376→    market: "",
   377→    sqft: String(WHAT_IF_DEFAULT_SQFT),
   378→    acreage: String(WHAT_IF_DEFAULT_ACREAGE),
   379→    label: "",
   380→  });
   381→
   382→  const sortedSubmarkets = useMemo(
   383→    () => [...(submarkets ?? [])].sort((a, b) => a.name.localeCompare(b.name)),
   384→    [submarkets]
   385→  );
   386→
   387→  const entries = useMemo(
   388→    () => [...starred.map(underwrite), ...whatIfs.map(underwrite)],
   389→    [starred, whatIfs]
   390→  );
   391→
   392→  const bestIncome = useMemo(() => {
   393→    const improved = entries.filter((m) => !m.isLand);
   394→    if (improved.length === 0) return null;
   395→    return improved.reduce((a, b) => (b.cashFlow > a.cashFlow ? b : a));
   396→  }, [entries]);
   397→  const bestEquity = useMemo(() => {
   398→    if (entries.length === 0) return null;
   399→    return entries.reduce((a, b) => (b.fv2046 > a.fv2046 ? b : a));
   400→  }, [entries]);
   401→
   402→  /* ---- What-if editor plumbing (all toasts fire in event handlers,
   403→     never inside state updaters — the round-6 bug class). The editor is
   404→     ADD-only as of round 8-a: every open resets to starred-derived
   405→     defaults, and the 3-entry limit is gated here so a 4th column can
   406→     never even open the form. ---- */
   407→
   408→  const canWhatIf = starred.length >= 2 && sortedSubmarkets.length > 0;
   409→
   410→  /** Median starred asking price, rounded to the nearest $5k. */
   411→  const defaultWhatIfPrice = useMemo(() => {
   412→    const prices = starred.map((l) => l.price).sort((a, b) => a - b);
   413→    if (prices.length === 0) return 450_000;
   414→    const mid = Math.floor(prices.length / 2);
   415→    const median =
   416→      prices.length % 2 === 1
   417→        ? prices[mid]
   418→        : (prices[mid - 1] + prices[mid]) / 2;
   419→    return Math.round(median / WHAT_IF_PRICE_STEP) * WHAT_IF_PRICE_STEP;
   420→  }, [starred]);
   421→
   422→  const handleEditorOpenChange = (next: boolean) => {
   423→    if (next) {
   424→      // Limit gate at open time (event-handler context, never an updater) —
   425→      // the toast fires once per click and the popover simply stays closed.
   426→      if (whatIfs.length >= WHAT_IF_MAX) {
   427→        toast({
   428→          title: "What-if limit reached",
   429→          description:
   430→            "Three hypotheticals at a time — remove one to add another.",
   431→        });
   432→        return;
   433→      }
   434→      const firstStarredSlug = starred[0]?.submarket?.slug ?? "";
   435→      const defaultSlug = sortedSubmarkets.some(
   436→        (s) => s.slug === firstStarredSlug
   437→      )
   438→        ? firstStarredSlug
   439→        : (sortedSubmarkets[0]?.slug ?? "");
   440→      setForm({
   441→        price: String(defaultWhatIfPrice),
   442→        type: "Single-Family",
   443→        market: defaultSlug,
   444→        sqft: String(WHAT_IF_DEFAULT_SQFT),
   445→        acreage: String(WHAT_IF_DEFAULT_ACREAGE),
   446→        label: "",
   447→      });
   448→    }
   449→    setEditorOpen(next);
   450→  };
   451→
   452→  const submitWhatIf = (e: FormEvent<HTMLFormElement>) => {
   453→    e.preventDefault();
   454→    // Defensive double-gate: the popover can't open at the limit, but a
   455→    // stray Enter (or a future programmatic submit) must not grow a 4th column.
   456→    if (whatIfs.length >= WHAT_IF_MAX) {
   457→      toast({
   458→        title: "What-if limit reached",
   459→        description:
   460→          "Three hypotheticals at a time — remove one to add another.",
   461→      });
   462→      return;
   463→    }
   464→    const parsed = Number(form.price);
   465→    if (!Number.isFinite(parsed) || parsed < WHAT_IF_MIN_PRICE) {
   466→      toast({
   467→        title: "Enter a target price first",
   468→        description: `The hypothetical needs a target price of at least ${fmtCurrency(
   469→          WHAT_IF_MIN_PRICE
   470→        )} — in ${fmtCurrency(WHAT_IF_PRICE_STEP)} steps — to underwrite.`,
   471→      });
   472→      return;
   473→    }
   474→    const market = sortedSubmarkets.find((s) => s.slug === form.market);
   475→    if (!market) {
   476→      toast({
   477→        title: "Pick a market first",
   478→        description:
   479→          "Choose the micro-market the hypothetical would trade in.",
   480→      });
   481→      return;
   482→    }
   483→    const price = Math.round(parsed / WHAT_IF_PRICE_STEP) * WHAT_IF_PRICE_STEP;
   484→    const sqft = clampWhatIfNumber(form.sqft, 0, WHAT_IF_MAX_SQFT);
   485→    const acreage = clampWhatIfNumber(form.acreage, 0, WHAT_IF_MAX_ACREAGE);
   486→    const label = form.label.trim() || uniqueAutoWhatIfLabel(price, whatIfs);
   487→    const submarket: ListingSubmarketSummary = {
   488→      slug: market.slug,
   489→      name: market.name,
   490→      state: market.state,
   491→      county: market.county,
   492→      regulatoryFramework: market.regulatoryFramework,
   493→      projectedCagr: market.projectedCagr,
   494→      depletionYear: market.depletionYear,
   495→      baselinePrice2026: market.baselinePrice2026,
   496→    };
   497→    const isLand = form.type === "Land Parcel" || sqft === 0;
   498→    const id = whatIfId(nextWhatIfSlot(whatIfs));
   499→    setWhatIfs((prev) => [
   500→      ...prev,
   501→      {
   502→        id,
   503→        submarketId: id,
   504→        title: label,
   505→        propertyType: form.type,
   506→        price,
   507→        acreage,
   508→        bedrooms: isLand ? 0 : 3,
   509→        bathrooms: isLand ? 0 : 2,
   510→        squareFeet: sqft,
   511→        zoningCode: "—",
   512→        description: "",
   513→        imageUrl: "",
   514→        featured: false,
   515→        status: "Hypothesis",
   516→        createdAt: new Date().toISOString(),
   517→        submarket,
   518→      },
   519→    ]);
   520→    setEditorOpen(false);
   521→    toast({
   522→      title: "What-if added to sheet",
   523→      description: `${label} · ${market.name} (${market.state}) — ${
   524→        whatIfs.length + 1
   525→      } of ${WHAT_IF_MAX} hypotheticals, underwritten with the same sheet as the ${starred.length} starred listings.`,
   526→    });
   527→  };
   528→
   529→  const removeWhatIf = (whatIf: PropertyListing) => {
   530→    setWhatIfs((prev) => prev.filter((w) => w.id !== whatIf.id));
   531→    const remaining = whatIfs.length - 1;
   532→    toast({
   533→      title: "What-if removed",
   534→      description: `${whatIf.title} was dropped — ${
   535→        remaining === 0
   536→          ? "only live listings remain."
   537→          : `${remaining} hypothetical${remaining === 1 ? "" : "s"} still on the sheet.`
   538→      }`,
   539→    });
   540→  };
   541→
   542→  /** Verdict pick title, tagged when ANY hypothetical wins a pick. */
   543→  const pickTitle = (m: CompareMetrics): string =>
   544→    isWhatIf(m.listing.id)
   545→      ? `${m.listing.title} (what-if)`
   546→      : m.listing.title;
   547→
   548→  /** Live preview backing the label field's placeholder. */
   549→  const formPrice = Number(form.price);
