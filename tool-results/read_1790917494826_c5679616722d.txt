    84→    84→export function CorridorMap({

    93→    93→  const [hovered, setHovered] = useState<string | null>(null);
    94→    94→  const [mode, setMode] = useState<MapMode>("cagr");

    96→    96→  /* ---------------- Depletion-timeline scrubber ---------------- */
    97→    97→  const [timelineYear, setTimelineYear] = useState(T0);
    98→    98→  const [playing, setPlaying] = useState(false);
    99→    99→  // Mirror the year in a ref so the rAF loop can capture its start value
   100→   100→  // without re-arming on every frame (exhaustive-deps stays clean).
   101→   101→  const yearRef = useRef(timelineYear);
   102→   102→  useEffect(() => {
   103→   103→    yearRef.current = timelineYear;
   104→   104→  }, [timelineYear]);
   105→   105→
   106→   106→  // prefers-reduced-motion — matched lazily after mount (matchMedia is
   107→   107→  // browser-only; reading it during render would break SSR/hydration) and
   108→   108→  // kept live via the change event, so a mid-session preference change
   109→   109→  // swaps the play loop in place.
   110→   110→  const [reducedMotion, setReducedMotion] = useState(false);
   111→   111→  useEffect(() => {
   112→   112→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   113→   113→    const sync = () => setReducedMotion(mq.matches);
   114→   114→    sync();
   115→   115→    mq.addEventListener("change", sync);
   116→   116→    return () => mq.removeEventListener("change", sync);
   117→   117→  }, []);
   118→   118→
   119→   119→  useEffect(() => {
   120→   120→    if (!playing) return;
   121→   121→    // Reduced motion: one discrete year per tick — no rAF interpolation,
   122→   122→    // no smooth tweening. Pause/reset/scrub behavior is unchanged.
   123→   123→    if (reducedMotion) {
   124→   124→      let year = yearRef.current;
   125→   125→      const id = window.setInterval(() => {
   126→   126→        year = Math.min(T1, year + 1);
   127→   127→        setTimelineYear(year);
   128→   128→        if (year >= T1) {
   129→   129→          setPlaying(false);
   130→   130→          window.clearInterval(id);
   131→   131→        }
   132→   132→      }, REDUCED_PLAY_MS_PER_YEAR);
   133→   133→      return () => window.clearInterval(id);
   134→   134→    }
   135→   135→    const startedAt = performance.now();
   136→   136→    const from = yearRef.current;
   137→   137→    let raf = 0;
   138→   138→    const tick = (now: number) => {
   139→   139→      const next = Math.min(T1, Math.round(from + (now - startedAt) / PLAY_MS_PER_YEAR));
   140→   140→      setTimelineYear(next);
   141→   141→      if (next >= T1) {
   142→   142→        setPlaying(false);
   143→   143→        return;
   144→   144→      }
   145→   145→      raf = requestAnimationFrame(tick);
   146→   146→    };
   147→   147→    raf = requestAnimationFrame(tick);
   148→   148→    return () => cancelAnimationFrame(raf);
   149→   149→  }, [playing, reducedMotion]);
   150→   150→
   151→   151→  const scrub = (year: number) => {
   152→   152→    setPlaying(false);
   153→   153→    setTimelineYear(year);
   154→   154→  };
   155→   155→  const togglePlay = () => {
   156→   156→    if (playing) {
   157→   157→      setPlaying(false);
   158→   158→      return;
   159→   159→    }
   160→   160→    // At the end of the reel, play restarts from the 2026 baseline.
   161→   161→    if (yearRef.current >= T1) setTimelineYear(T0);
   162→   162→    setPlaying(true);
   163→   163→  };
   164→   164→
   165→   165→  const spentCount = useMemo(
   166→   166→    () => submarkets.filter((s) => timelineYear >= s.depletionYear).length,
   167→   167→    [submarkets, timelineYear]
   168→   168→  );
   169→   169→
   170→   170→  /* Year-synced KPI readouts — every figure is recomputed from the
   171→   171→     submarket data at the scrub year (never hard-coded). */
   172→   172→  const kpis = useMemo(() => {
   173→   173→    // 1 · Median projected corridor price as of the scrub year.
   174→   174→    const projected = submarkets
   175→   175→      .map((s) => futureValue(s.baselinePrice2026, s.projectedCagr, timelineYear - T0))
   176→   176→      .sort((a, b) => a - b);
   177→   177→    const mid = Math.floor(projected.length / 2);
   178→   178→    const medianPrice =
   179→   179→      projected.length === 0
   180→   180→        ? 0
   181→   181→        : projected.length % 2 === 1
   182→   182→          ? projected[mid]
   183→   183→          : (projected[mid - 1] + projected[mid]) / 2;
   184→   184→    const priceDeltaPct =
   185→   185→      stats.regionalMedianPrice > 0
   186→   186→        ? (medianPrice / stats.regionalMedianPrice - 1) * 100
   187→   187→        : 0;
   188→   188→
   189→   189→    // 2 + 3 · Live reserve + next exhaustion, one pass over the corridor.
   190→   190→    let fullReserve = 0;
   191→   191→    let liveReserve = 0;
   192→   192→    let liveMarkets = 0;
   193→   193→    let next: { year: number; name: string; yearsOut: number; tierColor: string } | null =
   194→   194→      null;
   195→   195→    for (const s of submarkets) {
   196→   196→      const acresMid = netBuildableMid(s);
   197→   197→      fullReserve += acresMid;
   198→   198→      if (timelineYear < s.depletionYear) {
   199→   199→        liveReserve += acresMid;
   200→   200→        liveMarkets += 1;
   201→   201→        if (!next || s.depletionYear < next.year) {
   202→   202→          next = {
   203→   203→            year: s.depletionYear,
   204→   204→            name: s.name,
   205→   205→            yearsOut: s.depletionYear - timelineYear,
   206→   206→            tierColor: RUNWAY_TIER_STYLES[runwayTier(s.depletionYear - timelineYear)].color,
   207→   207→          };
   208→   208→        }
   209→   209→      }
   210→   210→    }
   211→   211→    const reservePct = fullReserve > 0 ? (liveReserve / fullReserve) * 100 : 0;
   212→   212→
   213→   213→    return { medianPrice, priceDeltaPct, liveReserve, liveMarkets, reservePct, next };
   214→   214→  }, [submarkets, stats.regionalMedianPrice, timelineYear]);
   215→   215→
   216→   216→  const dots = useMemo(
   217→   217→    () =>
   218→   218→      submarkets.map((s) => {
   219→   219→        const { px, py } = pos(s.mapX, s.mapY);
   220→   220→        const netMid = (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2;
   221→   221→        const r = 9 + netMid * 0.065;
   222→   222→        const tier = tierOf(s.projectedCagr);
   223→   223→        const label = LABEL_POS[s.slug] ?? { dx: 24, dy: 5, anchor: "start" as const };
   224→   224→        return { s, px, py, r, tier, netMid, label };
   225→   225→      }),
   226→   226→    [submarkets]
   227→   227→  );
   228→   228→
   229→   229→  const hoveredDot = dots.find((d) => d.s.slug === hovered) ?? null;
   230→   230→  const orShare = stats.orNetBuildable + stats.waNetBuildable > 0
   231→   231→    ? (stats.orNetBuildable / (stats.orNetBuildable + stats.waNetBuildable)) * 100
   232→   232→    : 50;
   233→   233→
   234→   234→  return (
   235→   235→    <div className="rounded-xl border bg-card p-4 shadow-sm sm:p-6">
   236→   236→      <div className="mb-4 flex flex-wrap items-start justify-between gap-3">
   237→   237→        <div>
   238→   238→          <MicroLabel>Regional Map Summary</MicroLabel>
   239→   239→          <p className="mt-1 text-sm text-muted-foreground">
   240→   240→            All 11 jurisdictions · dot size = net buildable acres ·{" "}
   241→   241→            {mode === "cagr"
   242→   242→              ? "color = 20-yr CAGR tier"
   243→   243→              : "color = raw-land runway remaining"}
   244→   244→            {timelineYear > T0 ? (
   245→   245→              <>
   246→   246→                {" · as of "}
   247→   247→                <span className="font-semibold tabular-nums text-foreground">
   248→   248→                  {timelineYear}
   249→   249→                </span>
   250→   250→              </>
   251→   251→            ) : null}
   252→   252→          </p>
   253→   253→        </div>
   254→   254→        {/* Lens toggle — segmented control */}
   255→   255→        <div
   256→   256→          role="radiogroup"
   257→   257→          aria-label="Map color lens"
   258→   258→          className="flex rounded-lg border bg-background p-0.5"
   259→   259→        >
   260→   260→          {MODES.map((m) => {
   261→   261→            const active = mode === m.id;
   262→   262→            return (
   263→   263→              <button
   264→   264→                key={m.id}
   265→   265→                type="button"
   266→   266→                role="radio"
   267→   267→                aria-checked={active}
   268→   268→                title={m.hint}
   269→   269→                onClick={() => setMode(m.id)}
   270→   270→                className={cn(
   271→   271→                  "inline-flex h-7 items-center gap-1.5 rounded-md px-2.5 text-[12px] font-semibold transition-all active:scale-[0.97]",
   272→   272→                  "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60 focus-visible:ring-offset-1 focus-visible:ring-offset-background",
   273→   273→                  active
   274→   274→                    ? "bg-zinc-900 text-white shadow-sm dark:bg-emerald-500 dark:text-zinc-950"
   275→   275→                    : "text-muted-foreground hover:text-foreground"
   276→   276→                )}
   277→   277→              >
   278→   278→                <m.icon className="h-3.5 w-3.5" aria-hidden />
   279→   279→                {m.label}
   280→   280→              </button>
   281→   281→            );
   282→   282→          })}
   283→   283→        </div>
   284→   284→      </div>
   285→   285→
   286→   286→      <div className="relative overflow-x-auto">
   287→   287→        <div className="relative min-w-[680px]">
   288→   288→          <svg
   289→   289→            viewBox={`0 0 ${VW} ${VH}`}
   290→   290→            className="h-auto w-full"
   291→   291→            role="img"
   292→   292→            aria-label="Map of the Columbia River Gorge corridor showing the 11 analyzed micro-markets on both sides of the river"
   293→   293→          >
   294→   294→            {/* State backdrops */}
   295→   295→            <rect x="0" y="0" width={VW} height="205" rx="10" className="fill-teal-500/[0.04] dark:fill-teal-400/[0.05]" />
   296→   296→            <rect x="0" y="255" width={VW} height={VH - 255} rx="10" className="fill-amber-500/[0.05] dark:fill-amber-400/[0.04]" />
   297→   297→
   298→   298→            <text x="28" y="36" className="fill-zinc-400 dark:fill-zinc-500" fontSize="17" fontWeight="700" letterSpacing="6">
   299→   299→              WASHINGTON
   300→   300→            </text>
   301→   301→            <text x="28" y={VH - 18} className="fill-zinc-400 dark:fill-zinc-500" fontSize="17" fontWeight="700" letterSpacing="6">
   302→   302→              OREGON
   303→   303→            </text>
   304→   304→
   305→   305→            {/* Highways */}
   306→   306→            <path
   307→   307→              d="M 15 300 C 160 285, 300 315, 450 300 C 600 285, 740 315, 985 300"

   341→   341→
   342→   342→            {/* Market dots */}

   345→   345→              const spent = timelineYear >= s.depletionYear;

   348→   348→              return (
   349→   349→                <g
