     1→"use client";
     2→

     4→import {
     5→  CartesianGrid,
     6→  Line,
     7→  LineChart,
     8→  ResponsiveContainer,
     9→  Tooltip,

    12→} from "recharts";

    14→import { Slider } from "@/components/ui/slider";
    15→import { cn } from "@/lib/utils";
    16→import {
    17→  INSURANCE_RATE,
    18→  PROPERTY_TAX_RATES,
    19→  RENTAL_RESERVE_RATE,
    20→  estimateMarketRent,
    21→  SEASONAL_RENT_BANDS,
    22→  type RentSeasonality,
    23→  fmtCurrency,
    24→  fmtPct,

    33→/* ---------------------------------------------------------------- */
    34→/* Financing Lab — amortized mortgage + carry-cost calculator         */
    35→/* embedded in every listing dossier.                                 */
    36→/* ---------------------------------------------------------------- */
    37→
    38→const TERMS = [15, 20, 30] as const;
    39→
    40→/** Investor rent postures applied on top of the market-rent heuristic. */
    41→const RENT_POSTURES = [
    42→  { id: "conservative", label: "Conservative", mult: 0.85 },
    43→  { id: "base", label: "Base", mult: 1 },
    44→  { id: "aggressive", label: "Aggressive", mult: 1.15 },
    45→] as const;
    46→type RentPosture = (typeof RENT_POSTURES)[number]["id"];
    47→
    48→/** Seasonality band order for the segmented control (round 8-e). */
    49→const SEASON_ORDER: RentSeasonality[] = ["lean", "annualized", "peak"];
    50→
    51→export function FinancingLab({ listing }: { listing: PropertyListing }) {
    52→  const isLand = listing.squareFeet === 0;
    53→  const defaultDown = isLand ? 35 : 20;
    54→
    55→  const [downPct, setDownPct] = useState(defaultDown);
    56→  const [rate, setRate] = useState(isLand ? 8.5 : 6.5);
    57→  const [term, setTerm] = useState<(typeof TERMS)[number]>(30);
    58→  const [scheduleOpen, setScheduleOpen] = useState(false);
    59→  const [posture, setPosture] = useState<RentPosture>("base");
    60→  const [seasonality, setSeasonality] = useState<RentSeasonality>("annualized");
    61→
    62→  const state = listing.submarket?.state ?? "OR";
    63→  const baseRent = useMemo(
    64→    () => estimateMarketRent(listing, { seasonality }),
    65→    [listing, seasonality]
    66→  );
    67→  const postureMult =
    68→    RENT_POSTURES.find((p) => p.id === posture)?.mult ?? 1;
    69→
    70→  const calc = useMemo(() => {
    71→    const price = Math.max(1, listing.price);
    72→    const down = (price * downPct) / 100;
    73→    const loan = price - down;
    74→    const pi = monthlyPayment(loan, rate, term);
    75→    const taxMonthly = (price * PROPERTY_TAX_RATES[state]) / 100 / 12;
    76→    const insMonthly = isLand ? 0 : (price * INSURANCE_RATE) / 12;
    77→    const totalMonthly = pi + taxMonthly + insMonthly;
    78→    const interest = totalInterest(loan, rate, term);
    79→
    80→    // Income lens: gross rent (market heuristic × investor posture) → NOI
    81→    // after an 8% operating reserve, then cash flow under the live note.
    82→    const grossAnnual =
    83→      (baseRent.dwelling * postureMult + baseRent.agriculture) * 12;
    84→    const noi = grossAnnual * (1 - RENTAL_RESERVE_RATE);
    85→
    86→    // 20-year equity runway: value compounds at the submarket CAGR while the
    87→    // amortizing balance burns down — the classic Gorge leveraged hold.
    88→    const cagr = listing.submarket?.projectedCagr ?? 5;
    89→    const value2046 = futureValue(price, cagr, 20);
    90→    const balance2046 = remainingBalance(loan, rate, term, 20 * 12);
    91→    const equity2046 = value2046 - balance2046;
    92→    const leverageMultiple =
    93→      down > 0 && equity2046 > 0 ? equity2046 / down : 0;
    94→
    95→    // Yearly amortization schedule + the note's "half-life": the first
    96→    // year-end balance below 50% of the original note.
    97→    const schedule: {
    98→      year: number;
    99→      principalY: number;
   100→      interestY: number;
   101→      cumInterest: number;
   102→      balance: number;
   103→      value: number;
   104→      halfWay: boolean;
   105→    }[] = [];
   106→    let prev = loan;
   107→    let cumInterest = 0;
   108→    let halfLifeYear: number | null = null;
   109→    for (let y = 1; y <= term; y++) {
   110→      const balance = remainingBalance(loan, rate, term, y * 12);
   111→      const principalY = prev - balance;
   112→      const interestY = pi * 12 - principalY;
   113→      cumInterest += interestY;
   114→      const halfWay =
   115→        halfLifeYear === null && balance <= loan / 2 && balance > 0;
   116→      if (halfWay) halfLifeYear = y;
   117→      schedule.push({
   118→        year: 2026 + y,
   119→        principalY,
   120→        interestY,
   121→        cumInterest,
   122→        balance,
   123→        value: futureValue(price, cagr, y),
   124→        halfWay,
   125→      });
   126→      prev = balance;
   127→    }
   128→
   129→    return {
   130→      down,
   131→      loan,
   132→      pi,
   133→      taxMonthly,
   134→      insMonthly,
   135→      totalMonthly,
   136→      interest,
   137→      value2046,
   138→      balance2046,
   139→      equity2046,
   140→      leverageMultiple,
   141→      cagr,
   142→      schedule,
   143→      halfLifeYear,
   144→      income: {
   145→        rentMonthly: baseRent.dwelling * postureMult + baseRent.agriculture,
   146→        agMonthly: baseRent.agriculture,
   147→        grossAnnual,
   148→        noi,
   149→        grossYield: (grossAnnual / price) * 100,
   150→        noiYield: (noi / price) * 100,
   151→        annualCarry: totalMonthly * 12,
   152→        annualCashFlow: noi - totalMonthly * 12,
   153→        cashOnCash: down > 0 ? (noi - totalMonthly * 12) / down : 0,
   154→      },
   155→    };
   156→  }, [listing, listing.price, listing.submarket?.projectedCagr, downPct, rate, term, state, isLand, baseRent, postureMult]);
   157→
   158→  const activeSeason = SEASONAL_RENT_BANDS[seasonality];
   159→
   160→  return (
   161→    <div className="rounded-lg border bg-background p-4">
   162→      <div className="flex items-center justify-between gap-2">
   163→        <div className="flex items-center gap-2">
   164→          <Calculator className="h-4 w-4 text-muted-foreground" aria-hidden />
   165→          <MicroLabel>Financing Lab</MicroLabel>
   166→        </div>
   167→        <span className="rounded-sm border bg-muted/60 px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   168→          {state === "WA" ? "WA carry costs" : "OR carry costs"}
   169→        </span>
   170→      </div>
   171→
   172→      {isLand ? (
   173→        <p className="mt-2.5 flex items-start gap-2 rounded-md border border-amber-500/30 bg-amber-500/[0.07] p-2.5 text-[12px] leading-relaxed text-amber-700 dark:text-amber-300">
   174→          <Info className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   175→          Raw-land paper: Gorge lenders typically demand 30–50% down on
   176→          shorter amortizations — the lab pre-loads 35% down at 8.5% to
   177→          reflect land-loan pricing.
   178→        </p>
   179→      ) : null}
   180→
   181→      {/* Inputs */}
   182→      <div className="mt-4 grid gap-4 sm:grid-cols-2">
   183→        <div>
   184→          <div className="mb-1.5 flex items-baseline justify-between">
   185→            <label htmlFor="fin-down" className="text-[12px] text-muted-foreground">
   186→              Down payment
   187→            </label>
   188→            <span className="text-[12.5px] font-semibold tabular-nums">
   189→              {downPct}% · {fmtCurrency(calc.down, { compact: true })}
   190→            </span>
   191→          </div>
   192→          <Slider
   193→            id="fin-down"
   194→            value={[downPct]}
   195→            onValueChange={(v) => setDownPct(v[0])}
   196→            min={5}
   197→            max={75}
   198→            step={1}
   199→            aria-label="Down payment percent"
   200→          />
   201→        </div>
   202→        <div>
   203→          <div className="mb-1.5 flex items-baseline justify-between">
   204→            <label htmlFor="fin-rate" className="text-[12px] text-muted-foreground">
   205→              Rate · APR
   206→            </label>
   207→            <span className="text-[12.5px] font-semibold tabular-nums">
   208→              {fmtPct(rate, 2)}
   209→            </span>
   210→          </div>
   211→          <Slider
   212→            id="fin-rate"
   213→            value={[rate]}
   214→            onValueChange={(v) => setRate(v[0])}
   215→            min={3}
   216→            max={12}
   217→            step={0.125}
   218→            aria-label="Interest rate"
   219→          />
   220→        </div>
   221→      </div>
   222→
   223→      <div className="mt-3.5 flex flex-wrap items-center gap-2">
   224→        <span className="text-[12px] text-muted-foreground">Term</span>
   225→        {TERMS.map((t) => (
   226→          <button
   227→            key={t}
   228→            type="button"
   229→            onClick={() => setTerm(t)}
   230→            aria-pressed={term === t}
   231→            className={cn(
   232→              "h-7 rounded-md border px-3 text-[12px] font-semibold tabular-nums transition-all active:scale-[0.97]",
   233→              term === t
   234→                ? "border-zinc-900 bg-zinc-900 text-white dark:border-emerald-500 dark:bg-emerald-500 dark:text-zinc-950"
   235→                : "bg-background text-muted-foreground hover:border-zinc-400 dark:hover:border-zinc-600"
   236→            )}
   237→          >
   238→            {t} yr
   239→          </button>
   240→        ))}
   241→      </div>
   242→
   243→      {/* Monthly stack */}
   244→      <div className="mt-4 rounded-lg border bg-card p-3.5">
   245→        <div className="flex items-baseline justify-between border-b border-border/60 pb-2.5">
   246→          <span className="text-[13px] font-semibold">Monthly carry</span>
   247→          <span className="text-xl font-semibold tabular-nums">
   248→            {fmtCurrency(Math.round(calc.totalMonthly))}
   249→          </span>
   250→        </div>
   251→        <div className="mt-2 space-y-1.5">
   252→          {[
   253→            {
   254→              label: "Principal & interest",
   255→              value: fmtCurrency(Math.round(calc.pi)),
   256→              note: `${fmtCurrency(calc.loan, { compact: true })} note`,
   257→            },
   258→            {
   259→              label: `Property tax · ${fmtPct(PROPERTY_TAX_RATES[state], 2)} eff.`,
   260→              value: fmtCurrency(Math.round(calc.taxMonthly)),
   261→              note: `${state === "WA" ? "Klickitat/Skamania" : "HR/Wasco"} blend`,
   262→            },
   263→            {
   264→              label: "Insurance",
   265→              value: calc.insMonthly > 0 ? fmtCurrency(Math.round(calc.insMonthly)) : "—",
   266→              note: isLand ? "n/a on unimproved land" : "structure heuristic",
   267→            },
   268→          ].map((row) => (
   269→            <div key={row.label} className="flex items-baseline justify-between gap-3">
   270→              <span className="text-[12px] text-muted-foreground">
   271→                {row.label}
   272→                <span className="ml-1.5 text-[10.5px] opacity-70">({row.note})</span>
   273→              </span>
   274→              <span className="text-[12.5px] font-medium tabular-nums">{row.value}</span>
   275→            </div>
   276→          ))}
   277→        </div>
   278→        <div className="mt-2.5 flex items-baseline justify-between border-t border-border/60 pt-2.5 text-[12px] text-muted-foreground">
   279→          <span className="inline-flex items-center gap-1.5">
   280→            <Landmark className="h-3.5 w-3.5" aria-hidden />
   281→            Lifetime interest · full {term}-yr note
   282→          </span>
   283→          <span className="font-medium tabular-nums text-rose-600 dark:text-rose-400">
   284→            {fmtCurrency(Math.round(calc.interest), { compact: true })}
   285→          </span>
   286→        </div>
   287→      </div>
   288→
   289→      {/* Income lens — rental stress-test */}
   290→      {!isLand ? (
   291→        <div className="mt-3 rounded-lg border bg-card p-3.5">
   292→          <div className="flex flex-wrap items-center justify-between gap-x-4 gap-y-2">
   293→            <div className="flex items-center gap-2">
   294→              <Warehouse className="h-4 w-4 text-muted-foreground" aria-hidden />
   295→              <MicroLabel>Income Lens · Rental Stress-Test</MicroLabel>
   296→            </div>
   297→            <div className="flex flex-wrap items-center gap-2.5">
   298→              {/* Seasonality band (round 8-e) — tourism-shaped demand */}
   299→              <div className="flex items-center gap-1.5">
   300→                <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   301→                  Season
   302→                </span>
   303→                <div
   304→                  role="radiogroup"
   305→                  aria-label="Rent seasonality band"
   306→                  className="flex rounded-md border bg-background p-0.5"
   307→                >
   308→                  {SEASON_ORDER.map((id) => {
   309→                    const band = SEASONAL_RENT_BANDS[id];
   310→                    const active = seasonality === id;
   311→                    return (
   312→                      <button
   313→                        key={id}
   314→                        type="button"
   315→                        role="radio"
   316→                        aria-checked={active}
   317→                        onClick={() => setSeasonality(id)}
   318→                        title={`${band.note} · rent × ${band.mult.toFixed(2)}`}
   319→                        className={cn(
   320→                          "h-6 rounded px-2 text-[11px] font-semibold transition-all active:scale-[0.97] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60",
   321→                          active
   322→                            ? "bg-zinc-900 text-white dark:bg-emerald-500 dark:text-zinc-950"
   323→                            : "text-muted-foreground hover:text-foreground"
   324→                        )}
   325→                      >
   326→                        {band.label}
   327→                      </button>
   328→                    );
   329→                  })}
   330→                </div>
   331→              </div>
   332→              {/* Investor posture on top of the market heuristic */}
   333→              <div className="flex items-center gap-1.5">
   334→                <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   335→                  Posture
   336→                </span>
   337→                <div
   338→                  role="radiogroup"
   339→                  aria-label="Rent posture"
   340→                  className="flex rounded-md border bg-background p-0.5"
   341→                >
   342→                  {RENT_POSTURES.map((p) => (
   343→                    <button
   344→                      key={p.id}
   345→                      type="button"
   346→                      role="radio"
   347→                      aria-checked={posture === p.id}
   348→                      onClick={() => setPosture(p.id)}
   349→                      title={`Market rent × ${p.mult.toFixed(2)}`}
   350→                      className={cn(
   351→                        "h-6 rounded px-2 text-[11px] font-semibold transition-all active:scale-[0.97] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60",
   352→                        posture === p.id
   353→                          ? "bg-zinc-900 text-white dark:bg-emerald-500 dark:text-zinc-950"
   354→                          : "text-muted-foreground hover:text-foreground"
   355→                      )}
   356→                    >
   357→                      {p.label}
   358→                    </button>
   359→                  ))}
   360→                </div>
   361→              </div>
   362→            </div>
   363→          </div>
   364→
   365→          {seasonality !== "annualized" ? (
   366→            <p className="mt-2 text-[10.5px] leading-snug text-muted-foreground">
   367→              <SunSnow className="mr-1 inline h-3 w-3 align-[-1px]" aria-hidden />
   368→              {activeSeason.label} band — {activeSeason.note} (rent ×{" "}
   369→              {activeSeason.mult.toFixed(2)}).
   370→            </p>
   371→          ) : null}
   372→
   373→          <div className="mt-2.5 flex items-baseline justify-between gap-3 border-b border-border/60 pb-2.5">
   374→            <span className="text-[12px] text-muted-foreground">
   375→              Est. market rent
   376→              <span className="ml-1.5 text-[10.5px] opacity-70">({baseRent.note})</span>
   377→            </span>
   378→            <span className="text-[15px] font-semibold tabular-nums">
   379→              {fmtCurrency(Math.round(calc.income.rentMonthly))}
   380→              <span className="text-[11px] font-normal text-muted-foreground">/mo</span>
   381→            </span>
   382→          </div>
   383→
   384→          <div className="mt-2 grid grid-cols-2 gap-2 sm:grid-cols-4">
   385→            {[
   386→              {
   387→                label: "Gross yield",
   388→                value: fmtPct(calc.income.grossYield),
   389→                sub: "rent ÷ price",
   390→                tone: "default" as const,
   391→              },
   392→              {
   393→                label: "NOI yield",
   394→                value: fmtPct(calc.income.noiYield),
   395→                sub: `after ${fmtPct(RENTAL_RESERVE_RATE * 100, 0)} reserve`,
   396→                tone: calc.income.noiYield >= rate ? ("emerald" as const) : ("default" as const),
   397→              },
   398→              {
   399→                label: "Annual cash flow",
   400→                value: fmtCurrency(Math.round(calc.income.annualCashFlow), { compact: true }),
   401→                sub: "NOI − note carry",
   402→                tone: calc.income.annualCashFlow >= 0 ? ("emerald" as const) : ("rose" as const),
   403→              },
   404→              {
   405→                label: "Cash-on-cash",
   406→                value: `${calc.income.cashOnCash >= 0 ? "" : "−"}${Math.abs(calc.income.cashOnCash * 100).toFixed(1)}%`,
   407→                sub: "on the down stroke",
   408→                tone: calc.income.cashOnCash >= 0 ? ("emerald" as const) : ("rose" as const),
   409→              },
   410→            ].map((m) => (
   411→              <div
   412→                key={m.label}
   413→                className={cn(
   414→                  "rounded-md border bg-background p-2.5 text-center",
   415→                  m.tone === "emerald" && "border-emerald-500/30 bg-emerald-500/[0.05]",
   416→                  m.tone === "rose" && "border-rose-500/30 bg-rose-500/[0.05]"
   417→                )}
   418→              >
   419→                <p className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   420→                  {m.label}
   421→                </p>
   422→                <p
   423→                  className={cn(
   424→                    "mt-0.5 text-[13.5px] font-semibold tabular-nums",
   425→                    m.tone === "emerald" && "text-emerald-600 dark:text-emerald-400",
   426→                    m.tone === "rose" && "text-rose-600 dark:text-rose-400"
   427→                  )}
   428→                >
   429→                  {m.value}
   430→                </p>
   431→                <p className="mt-0.5 text-[10px] text-muted-foreground">{m.sub}</p>
   432→              </div>
   433→            ))}
   434→          </div>
   435→
   436→          <p className="mt-2.5 border-t border-border/60 pt-2.5 text-[11.5px] leading-relaxed text-muted-foreground">
   437→            {calc.income.annualCashFlow >= 0 ? (
   438→              <>
   439→                At {fmtPct(calc.income.noiYield)} NOI yield vs a {fmtPct(rate, 2)} note,
   440→                the property carries itself —{" "}
   441→                <span className="font-semibold text-emerald-600 dark:text-emerald-400">
   442→                  {fmtCurrency(Math.round(calc.income.annualCashFlow), { compact: true })} / yr
   443→                </span>{" "}
   444→                of cash left after the full monthly stack.
   445→              </>
   446→            ) : (
   447→              <>
   448→                Negative leverage: the {fmtPct(rate, 2)} note outruns the{" "}
   449→                {fmtPct(calc.income.noiYield)} NOI yield by{" "}
   450→                <span className="font-semibold text-rose-600 dark:text-rose-400">
   451→                  {fmtCurrency(Math.round(-calc.income.annualCashFlow), { compact: true })} / yr
   452→                </span>{" "}
   453→                — the bet rests on the {fmtPct(calc.cagr)} appreciation engine, not rent.
   454→              </>
   455→            )}
   456→          </p>
   457→        </div>
   458→      ) : null}
   459→
   460→      {/* 2046 equity runway */}
   461→      <div className="mt-3 rounded-lg border border-emerald-500/25 bg-emerald-500/[0.05] p-3.5">
   462→        <div className="flex items-center gap-2">
   463→          <PiggyBank className="h-4 w-4 text-emerald-600 dark:text-emerald-400" aria-hidden />
   464→          <MicroLabel className="text-emerald-700 dark:text-emerald-300">
   465→            20-Year Equity Runway · {listing.submarket?.name ?? "corridor"} CAGR {fmtPct(calc.cagr)}
   466→          </MicroLabel>
   467→        </div>
   468→        <div className="mt-2.5 grid grid-cols-3 gap-2 text-center">
   469→          <div>
   470→            <p className="text-[10.5px] uppercase tracking-wide text-muted-foreground">
   471→              Value 2046
   472→            </p>
   473→            <p className="mt-0.5 text-[13.5px] font-semibold tabular-nums">
   474→              {fmtCurrency(calc.value2046, { compact: true })}
   475→            </p>
   476→          </div>
   477→          <div>
   478→            <p className="text-[10.5px] uppercase tracking-wide text-muted-foreground">
   479→              Loan balance
   480→            </p>
   481→            <p className="mt-0.5 text-[13.5px] font-semibold tabular-nums text-muted-foreground">
   482→              −{fmtCurrency(calc.balance2046, { compact: true })}
   483→            </p>
   484→          </div>
   485→          <div>
   486→            <p className="text-[10.5px] uppercase tracking-wide text-muted-foreground">
   487→              Equity
   488→            </p>
   489→            <p className="mt-0.5 text-[13.5px] font-semibold tabular-nums text-emerald-600 dark:text-emerald-400">
   490→              {fmtCurrency(calc.equity2046, { compact: true })}
   491→            </p>
   492→          </div>
   493→        </div>
   494→        {calc.leverageMultiple > 0 ? (
   495→          <p className="mt-2.5 border-t border-emerald-500/20 pt-2.5 text-[12px] leading-relaxed text-muted-foreground">
   496→            On a {fmtCurrency(calc.down, { compact: true })} down stroke, that is a{" "}
