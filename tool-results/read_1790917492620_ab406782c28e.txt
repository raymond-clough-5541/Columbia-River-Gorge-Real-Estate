   129→   129→function underwrite(l: PropertyListing): CompareMetrics {
   130→   130→  const isLand = l.propertyType === "Land Parcel" || l.squareFeet === 0;
   131→   131→  const price = Math.max(1, l.price);
   132→   132→  const downPct = isLand ? LAND_DOWN_PCT : SHEET_DOWN_PCT;
   133→   133→  const rate = isLand ? LAND_RATE : SHEET_RATE;
   134→   134→  const down = (price * downPct) / 100;
   135→   135→  const loan = price - down;
   136→   136→  const state = l.submarket?.state ?? "OR";
   137→   137→  const pi = monthlyPayment(loan, rate, SHEET_TERM);
   138→   138→  const tax = (price * PROPERTY_TAX_RATES[state]) / 100 / 12;
   139→   139→  const ins = isLand ? 0 : (price * INSURANCE_RATE) / 12;
   140→   140→  const monthlyCarry = pi + tax + ins;
   141→   141→
   142→   142→  const rent = estimateMarketRent(l); // market-indexed per round 6
   143→   143→  const grossAnnual = (rent.dwelling + rent.agriculture) * 12;
   144→   144→  const noi = grossAnnual * (1 - RENTAL_RESERVE_RATE);
   145→   145→  const cagr = l.submarket?.projectedCagr ?? 5;
   146→   146→  const fv2046 = futureValue(price, cagr, 20);
   147→   147→
   148→   148→  return {
   149→   149→    listing: l,
   150→   150→    isLand,
   151→   151→    loan,
   152→   152→    monthlyCarry,
   153→   153→    rentMonthly: rent.dwelling + rent.agriculture,
   154→   154→    grossYield: isLand ? 0 : (grossAnnual / price) * 100,
   155→   155→    noiYield: isLand ? 0 : (noi / price) * 100,
   156→   156→    cashFlow: noi - monthlyCarry * 12,
   157→   157→    cashOnCash: down > 0 ? (noi - monthlyCarry * 12) / down : 0,
   158→   158→    fv2046,
   159→   159→    multiple: Math.pow(1 + cagr / 100, 20),
   160→   160→    pricePerSqft: !isLand && l.squareFeet > 0 ? price / l.squareFeet : null,
   161→   161→    runway: runwayYears(l.submarket?.depletionYear ?? 2046),
   162→   162→  };
   163→   163→}

   274→   274→/** Pick the winning metric entry for a row (null when informational). */
   275→   275→function winnerOf(
   276→   276→  row: RowDef,
   277→   277→  entries: CompareMetrics[]
   278→   278→): CompareMetrics | null {
   279→   279→  if (!row.better || entries.length < 2) return null;
   280→   280→  const scored = entries.filter((m) => !row.na?.(m));
   281→   281→  if (scored.length < 2) return null;
   282→   282→  const val = (m: CompareMetrics) => {
   283→   283→    switch (row.key) {
   284→   284→      case "price": return m.listing.price;
   285→   285→      case "sqft": return m.pricePerSqft ?? Infinity;
   286→   286→      case "carry": return m.monthlyCarry;
   287→   287→      case "rent": return m.rentMonthly;
   288→   288→      case "gross": return m.grossYield;
   289→   289→      case "noi": return m.noiYield;
   290→   290→      case "cashflow": return m.cashFlow;
   291→   291→      case "coc": return m.cashOnCash;
   292→   292→      case "fv": return m.fv2046;
   293→   293→      case "mult": return m.multiple;
   294→   294→      case "runway": return m.runway;
   295→   295→      default: return 0;
   296→   296→    }
   297→   297→  };
   298→   298→  const sorted = [...scored].sort((a, b) =>
   299→   299→    row.better === "high" ? val(b) - val(a) : val(a) - val(b)
   300→   300→  );
   301→   301→  const best = sorted[0];
   302→   302→  // Ties → both win (highlight stays honest).
   303→   303→  return val(best) === val(sorted[1]) ? null : best;
   304→   304→}

   306→   306→export function CompareSheet({
   307→   307→  listings,
   308→   308→  open,
   309→   309→  onOpenChange,
   310→   310→  onOpenListing,
   311→   311→  submarkets,
   312→   312→}: {
   313→   313→  listings: PropertyListing[];
   314→   314→  open: boolean;
   315→   315→  onOpenChange: (open: boolean) => void;
   316→   316→  onOpenListing?: (listing: PropertyListing) => void;
   317→   317→  /** Real micro-markets the what-if editor can underwrite in. */
   318→   318→  submarkets?: Submarket[];
   319→   319→}) {
   320→   320→  const watchlist = useWatchlist();
   321→   321→  const { toast } = useToast();
   322→   322→
   323→   323→  // Star order = the user's shortlist order.
   324→   324→  const starred = useMemo(
   325→   325→    () =>
   326→   326→      watchlist
   327→   327→        .map((id) => listings.find((l) => l.id === id))
   328→   328→        .filter((l): l is PropertyListing => Boolean(l)),
   329→   329→    [watchlist, listings]
   330→   330→  );
   331→   331→  /* ---- Hypothetical what-if column (round 7-b) ----
   332→   332→     At most ONE synthetic listing. The state lives here — the sheet stays
   333→   333→     mounted across open/close via CompareTrigger — so the what-if survives
   334→   334→     closing and reopening the sheet for the whole session. */
   335→   335→  const [whatIf, setWhatIf] = useState<PropertyListing | null>(null);
   336→   336→  const [editorOpen, setEditorOpen] = useState(false);
   337→   337→  const [form, setForm] = useState<WhatIfForm>({
   338→   338→    price: "",
   339→   339→    type: "Single-Family",
   340→   340→    market: "",
   341→   341→    sqft: String(WHAT_IF_DEFAULT_SQFT),
   342→   342→    acreage: String(WHAT_IF_DEFAULT_ACREAGE),
   343→   343→    label: "",
   344→   344→  });
   345→   345→
   346→   346→  const sortedSubmarkets = useMemo(
   347→   347→    () => [...(submarkets ?? [])].sort((a, b) => a.name.localeCompare(b.name)),
   348→   348→    [submarkets]
   349→   349→  );
   350→   350→
   351→   351→  const entries = useMemo(() => {
   352→   352→    const base = starred.map(underwrite);
   353→   353→    return whatIf ? [...base, underwrite(whatIf)] : base;
   354→   354→  }, [starred, whatIf]);
   355→   355→
   356→   356→  const bestIncome = useMemo(() => {
   357→   357→    const improved = entries.filter((m) => !m.isLand);
   358→   358→    if (improved.length === 0) return null;
   359→   359→    return improved.reduce((a, b) => (b.cashFlow > a.cashFlow ? b : a));
   360→   360→  }, [entries]);
   361→   361→  const bestEquity = useMemo(() => {
   362→   362→    if (entries.length === 0) return null;
   363→   363→    return entries.reduce((a, b) => (b.fv2046 > a.fv2046 ? b : a));
   364→   364→  }, [entries]);
   365→   365→
   366→   366→  /* ---- What-if editor plumbing (all toasts fire in event handlers,
   367→   367→     never inside state updaters — the round-6 bug class). ---- */
   368→   368→
   369→   369→  const canWhatIf = starred.length >= 2 && sortedSubmarkets.length > 0;
   370→   370→
   371→   371→  /** Median starred asking price, rounded to the nearest $5k. */
   372→   372→  const defaultWhatIfPrice = useMemo(() => {
   373→   373→    const prices = starred.map((l) => l.price).sort((a, b) => a - b);
   374→   374→    if (prices.length === 0) return 450_000;
   375→   375→    const mid = Math.floor(prices.length / 2);
   376→   376→    const median =
   377→   377→      prices.length % 2 === 1
   378→   378→        ? prices[mid]
   379→   379→        : (prices[mid - 1] + prices[mid]) / 2;
   380→   380→    return Math.round(median / WHAT_IF_PRICE_STEP) * WHAT_IF_PRICE_STEP;
   381→   381→  }, [starred]);
   382→   382→
   383→   383→  const handleEditorOpenChange = (next: boolean) => {
   384→   384→    if (next) {
   385→   385→      const firstStarredSlug = starred[0]?.submarket?.slug ?? "";
   386→   386→      const defaultSlug = sortedSubmarkets.some(
   387→   387→        (s) => s.slug === firstStarredSlug
   388→   388→      )
   389→   389→        ? firstStarredSlug
   390→   390→        : (sortedSubmarkets[0]?.slug ?? "");
   391→   391→      setForm({
   392→   392→        price: whatIf ? String(whatIf.price) : String(defaultWhatIfPrice),
   393→   393→        type: whatIf ? whatIf.propertyType : "Single-Family",
   394→   394→        market: whatIf?.submarket?.slug ?? defaultSlug,
   395→   395→        sqft: whatIf ? String(whatIf.squareFeet) : String(WHAT_IF_DEFAULT_SQFT),
   396→   396→        acreage: whatIf
   397→   397→          ? String(whatIf.acreage)
   398→   398→          : String(WHAT_IF_DEFAULT_ACREAGE),
   399→   399→        label:
   400→   400→          whatIf && !WHAT_IF_AUTO_LABEL.test(whatIf.title) ? whatIf.title : "",
   401→   401→      });
   402→   402→    }
   403→   403→    setEditorOpen(next);
   404→   404→  };
   405→   405→
   406→   406→  const submitWhatIf = (e: FormEvent<HTMLFormElement>) => {
   407→   407→    e.preventDefault();
   408→   408→    const parsed = Number(form.price);
   409→   409→    if (!Number.isFinite(parsed) || parsed < WHAT_IF_MIN_PRICE) {
   410→   410→      toast({
   411→   411→        title: "Enter a target price first",
   412→   412→        description: `The hypothetical needs a target price of at least ${fmtCurrency(
   413→   413→          WHAT_IF_MIN_PRICE
   414→   414→        )} — in ${fmtCurrency(WHAT_IF_PRICE_STEP)} steps — to underwrite.`,
   415→   415→      });
   416→   416→      return;
   417→   417→    }
   418→   418→    const market = sortedSubmarkets.find((s) => s.slug === form.market);
   419→   419→    if (!market) {
   420→   420→      toast({
   421→   421→        title: "Pick a market first",
   422→   422→        description:
   423→   423→          "Choose the micro-market the hypothetical would trade in.",
   424→   424→      });
   425→   425→      return;
   426→   426→    }
   427→   427→    const price = Math.round(parsed / WHAT_IF_PRICE_STEP) * WHAT_IF_PRICE_STEP;
   428→   428→    const sqft = clampWhatIfNumber(form.sqft, 0, WHAT_IF_MAX_SQFT);
   429→   429→    const acreage = clampWhatIfNumber(form.acreage, 0, WHAT_IF_MAX_ACREAGE);
   430→   430→    const label = form.label.trim() || autoWhatIfLabel(price);
   431→   431→    const submarket: ListingSubmarketSummary = {
   432→   432→      slug: market.slug,
   433→   433→      name: market.name,
   434→   434→      state: market.state,
   435→   435→      county: market.county,
   436→   436→      regulatoryFramework: market.regulatoryFramework,
   437→   437→      projectedCagr: market.projectedCagr,
   438→   438→      depletionYear: market.depletionYear,
   439→   439→      baselinePrice2026: market.baselinePrice2026,
   440→   440→    };
   441→   441→    const isLand = form.type === "Land Parcel" || sqft === 0;
   442→   442→    setWhatIf({
   443→   443→      id: WHAT_IF_ID,
   444→   444→      submarketId: WHAT_IF_ID,
   445→   445→      title: label,
   446→   446→      propertyType: form.type,
   447→   447→      price,
   448→   448→      acreage,
   449→   449→      bedrooms: isLand ? 0 : 3,
   450→   450→      bathrooms: isLand ? 0 : 2,
   451→   451→      squareFeet: sqft,
   452→   452→      zoningCode: "—",
   453→   453→      description: "",
   454→   454→      imageUrl: "",
   455→   455→      featured: false,
   456→   456→      status: "Hypothesis",
   457→   457→      createdAt: new Date().toISOString(),
   458→   458→      submarket,
   459→   459→    });
   460→   460→    setEditorOpen(false);
