   109→   109→
   110→   110→export function MatrixView({
   111→   111→  submarkets,
   112→   112→  navigate,
   113→   113→}: {
   114→   114→  submarkets: Submarket[];
   115→   115→  navigate: NavigateFn;
   116→   116→}) {
   117→   117→  const [stateFilter, setStateFilter] = useState<"all" | "OR" | "WA">("all");
   118→   118→  const [jurisdictionFilter, setJurisdictionFilter] = useState<
   119→   119→    "all" | "Incorporated City" | "Unincorporated Urban Area"
   120→   120→  >("all");
   121→   121→  const [sortCol, setSortCol] = useState<SortCol>("net");
   122→   122→  const [sortDir, setSortDir] = useState<SortDir>("desc");
   123→   123→  const [detailSlug, setDetailSlug] = useState<string | null>(null);
   124→   124→  const [visibleCols, setVisibleCols] =
   125→   125→    useState<Record<ColumnKey, boolean>>(DEFAULT_VISIBLE);
   126→   126→  const [pins, setPins] = useState<string[]>([]);
   127→   127→
   128→   128→  const togglePin = (slug: string) => {
   129→   129→    // Event-time logic only — never inside a state updater (that would
   130→   130→    // update the Toaster while MatrixView renders → React warning).
   131→   131→    if (pins.includes(slug)) {
   132→   132→      setPins((prev) => prev.filter((s) => s !== slug));
   133→   133→      return;
   134→   134→    }
   135→   135→    if (pins.length >= MAX_PINS) {
   136→   136→      toast({
   137→   137→        title: "Pin limit reached",
   138→   138→        description: `Up to ${MAX_PINS} jurisdictions can stay pinned — unpin one first.`,
   139→   139→      });
   140→   140→      return;
   141→   141→    }
   142→   142→    setPins((prev) => (prev.includes(slug) ? prev : [...prev, slug]));
   143→   143→  };
   144→   144→
   145→   145→  const visibleCount = useMemo(
   146→   146→    () => ALL_COLUMNS.filter((c) => visibleCols[c]).length,
   147→   147→    [visibleCols]
   148→   148→  );
   149→   149→  const toggleCol = (col: ColumnKey) =>
   150→   150→    setVisibleCols((v) => ({ ...v, [col]: !v[col] }));
   151→   151→
   152→   152→  const rows = useMemo(() => {
   153→   153→    const mid = (s: Submarket) =>
   154→   154→      (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2;
   155→   155→    const forecastMid = (s: Submarket) =>
   156→   156→      (s.projectedPrice2046Min + s.projectedPrice2046Max) / 2;
   157→   157→
   158→   158→    const value = (s: Submarket): number => {
   159→   159→      switch (sortCol) {
   160→   160→        case "net":
   161→   161→          return mid(s);
   162→   162→        case "baseline":
   163→   163→          return s.baselinePrice2026;
   164→   164→        case "cagr":
   165→   165→          return s.projectedCagr;
   166→   166→        case "forecast":
   167→   167→          return forecastMid(s);
   168→   168→        case "footprint":
   169→   169→          return s.totalFootprintAcres;
   170→   170→        case "dom":
   171→   171→          return (s.daysOnMarketMin + s.daysOnMarketMax) / 2;
   172→   172→      }
   173→   173→    };
   174→   174→
   175→   175→    return submarkets
   176→   176→      .filter(
   177→   177→        (s) =>
   178→   178→          (stateFilter === "all" || s.state === stateFilter) &&
   179→   179→          (jurisdictionFilter === "all" || s.jurisdictionType === jurisdictionFilter)
   180→   180→      )
   181→   181→      .sort((a, b) =>
   182→   182→        sortDir === "asc" ? value(a) - value(b) : value(b) - value(a)
   183→   183→      );
   184→   184→  }, [submarkets, stateFilter, jurisdictionFilter, sortCol, sortDir]);
   185→   185→
   186→   186→  const filteredNet = rows.reduce(
   187→   187→    (a, s) => a + (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2,
   188→   188→    0
   189→   189→  );
   190→   190→  const filteredCagr =
   191→   191→    rows.length > 0
   192→   192→      ? rows.reduce((a, s) => a + s.projectedCagr, 0) / rows.length
   193→   193→      : 0;
   194→   194→
   195→   195→  const detail = submarkets.find((s) => s.slug === detailSlug) ?? null;
   196→   196→
   197→   197→  /** Pinned jurisdictions float above the fold, in pin order. */
   198→   198→  const orderedRows = useMemo(() => {
   199→   199→    const pinnedRows = pins
   200→   200→      .map((slug) => rows.find((r) => r.slug === slug))
   201→   201→      .filter((r): r is Submarket => Boolean(r));
   202→   202→    const rest = rows.filter((r) => !pins.includes(r.slug));
   203→   203→    return [...pinnedRows, ...rest];
   204→   204→  }, [rows, pins]);
   205→   205→
   206→   206→  const { toast } = useToast();
   207→   207→
   208→   208→  const exportCsv = () => {
   209→   209→    const headers = [
   210→   210→      "Market",
   211→   211→      "State",
   212→   212→      "County",
   213→   213→      "Jurisdiction Type",
   214→   214→      "Regulatory Framework",
   215→   215→      "UGB/UGA Footprint (ac)",
   216→   216→      "Gross Vacant (ac)",
   217→   217→      "Net Buildable Min (ac)",
   218→   218→      "Net Buildable Max (ac)",
   219→   219→      "Net Buildable Mid (ac)",
   220→   220→      "2026 Baseline Median ($)",
   221→   221→      "Price/SqFt Min ($)",
   222→   222→      "Price/SqFt Max ($)",
   223→   223→      "Days on Market Min",
   224→   224→      "Days on Market Max",
   225→   225→      "Projected 20-yr CAGR (%)",
   226→   226→      "2046 Forecast Min ($)",
   227→   227→      "2046 Forecast Max ($)",
   228→   228→      "Est. Raw-Land Depletion Year",
   229→   229→      "Water Purveyor",
   230→   230→      "Wastewater System",
   231→   231→      "Primary Constraints & WUI Risk",
   232→   232→    ];
   233→   233→    const data: CsvCell[][] = orderedRows.map((s) => [
   234→   234→      s.name,
   235→   235→      s.state,
   236→   236→      s.county,
   237→   237→      s.jurisdictionType,
   238→   238→      s.regulatoryFramework,
   239→   239→      s.totalFootprintAcres,
   240→   240→      s.grossVacantAcres,
   241→   241→      s.netBuildableAcresMin,
   242→   242→      s.netBuildableAcresMax,
   243→   243→      Math.round(((s.netBuildableAcresMin + s.netBuildableAcresMax) / 2) * 10) / 10,
   244→   244→      s.baselinePrice2026,
   245→   245→      Math.round(s.pricePerSqftMin),
   246→   246→      Math.round(s.pricePerSqftMax),
   247→   247→      s.daysOnMarketMin,
   248→   248→      s.daysOnMarketMax,
   249→   249→      s.projectedCagr,
   250→   250→      s.projectedPrice2046Min,
   251→   251→      s.projectedPrice2046Max,
   252→   252→      s.depletionYear,
   253→   253→      s.waterPurveyor,
   254→   254→      s.wastewaterSystem,
   255→   255→      s.primaryConstraints,
   256→   256→    ]);
   257→   257→    downloadCsv(
   258→   258→      `crgnsa-master-matrix-${timestampSuffix()}.csv`,
   259→   259→      toCsv(headers, data)
   260→   260→    );
   261→   261→    toast({
   262→   262→      title: "Master Matrix exported",
   263→   263→      description: `${rows.length} jurisdiction${rows.length === 1 ? "" : "s"} · current filters and sort order applied.`,
   264→   264→    });
   265→   265→  };
   266→   266→
   267→   267→  const toggleSort = (col: SortCol) => {
   268→   268→    if (col === sortCol) {
   269→   269→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   270→   270→    } else {
   271→   271→      setSortCol(col);
   272→   272→      setSortDir("desc");
   273→   273→    }
   274→   274→  };
   275→   275→
   276→   276→  return (
   277→   277→    <div className="mx-auto w-full max-w-7xl px-4 pb-16 pt-10 sm:px-6 lg:px-8">
   278→   278→      <SectionHeader
   279→   279→        eyebrow="Master Real Estate & Land Inventory Matrix"
   280→   280→        title="All 11 jurisdictions, one sortable ledger"
   281→   281→        description="Filter by state and jurisdiction type, sort any column, and click a row for the infrastructure dossier — water purveyors, wastewater capacity, and fire-risk constraints that gate every buildable acre."
   282→   282→        action={
   283→   283→          <div className="flex items-center gap-2">
   284→   284→            <DropdownMenu>
   285→   285→              <DropdownMenuTrigger asChild>
   286→   286→                <button
   287→   287→                  type="button"
   288→   288→                  aria-label="Toggle column visibility"
   289→   289→                  className="inline-flex h-9 items-center gap-2 rounded-lg border bg-card px-3.5 text-[13px] font-medium shadow-sm transition-all hover:border-zinc-400 hover:shadow dark:hover:border-zinc-600 active:scale-[0.98]"
   290→   290→                >
   291→   291→                  <Columns3 className="h-3.5 w-3.5" aria-hidden />
   292→   292→                  Columns
   293→   293→                  <span className="rounded-sm bg-muted px-1.5 text-[11px] font-semibold tabular-nums">
   294→   294→                    {visibleCount}/{ALL_COLUMNS.length}
   295→   295→                  </span>
   296→   296→                </button>
   297→   297→              </DropdownMenuTrigger>
   298→   298→              <DropdownMenuContent align="end" className="w-56">
   299→   299→                <DropdownMenuLabel className="text-[12px] uppercase tracking-wider">
   300→   300→                  Visible columns
   301→   301→                </DropdownMenuLabel>
   302→   302→                <DropdownMenuSeparator />
   303→   303→                {ALL_COLUMNS.map((col) => (
   304→   304→                  <DropdownMenuCheckboxItem
   305→   305→                    key={col}
   306→   306→                    checked={visibleCols[col]}
   307→   307→                    onCheckedChange={() => toggleCol(col)}
   308→   308→                    onSelect={(e) => e.preventDefault()}
   309→   309→                    className="text-[13px]"
   310→   310→                  >
   311→   311→                    {COLUMN_LABELS[col]}
   312→   312→                  </DropdownMenuCheckboxItem>
   313→   313→                ))}
   314→   314→              </DropdownMenuContent>
   315→   315→            </DropdownMenu>
   316→   316→            <button
   317→   317→              type="button"
   318→   318→              onClick={exportCsv}
   319→   319→              disabled={rows.length === 0}
   320→   320→              className="inline-flex h-9 items-center gap-2 rounded-lg border bg-card px-4 text-[13px] font-medium shadow-sm transition-all hover:border-zinc-400 hover:shadow disabled:cursor-not-allowed disabled:opacity-50 dark:hover:border-zinc-600 active:scale-[0.98]"
   321→   321→            >
   322→   322→              <Download className="h-3.5 w-3.5" aria-hidden />
   323→   323→              Export CSV
   324→   324→            </button>
   325→   325→          </div>
   326→   326→        }
   327→   327→      />
   328→   328→
   329→   329→      {/* Filter + summary bar */}
   330→   330→      <div className="mb-5 flex flex-wrap items-center gap-4 rounded-xl border bg-card p-4 shadow-sm">
   331→   331→        <div className="flex items-center gap-3">
   332→   332→          <MicroLabel className="hidden sm:block">State</MicroLabel>
   333→   333→          <ToggleGroup
   334→   334→            type="single"
   335→   335→            value={stateFilter}
   336→   336→            onValueChange={(v) => {
   337→   337→              if (v) setStateFilter(v as "all" | "OR" | "WA");
   338→   338→            }}
   339→   339→            className="gap-1"
   340→   340→          >
   341→   341→            <ToggleGroupItem value="all" className="h-8 px-3 text-[13px]">All</ToggleGroupItem>
   342→   342→            <ToggleGroupItem value="OR" className="h-8 px-3 text-[13px]">Oregon</ToggleGroupItem>
   343→   343→            <ToggleGroupItem value="WA" className="h-8 px-3 text-[13px]">Washington</ToggleGroupItem>
   344→   344→          </ToggleGroup>
   345→   345→        </div>
   346→   346→        <div className="flex items-center gap-3">
   347→   347→          <MicroLabel className="hidden sm:block">Jurisdiction</MicroLabel>
   348→   348→          <ToggleGroup
   349→   349→            type="single"
   350→   350→            value={jurisdictionFilter}
   351→   351→            onValueChange={(v) => {
   352→   352→              if (v)
   353→   353→                setJurisdictionFilter(
   354→   354→                  v as "all" | "Incorporated City" | "Unincorporated Urban Area"
   355→   355→                );
   356→   356→            }}
   357→   357→            className="gap-1"
   358→   358→          >
   359→   359→            <ToggleGroupItem value="all" className="h-8 px-3 text-[13px]">All</ToggleGroupItem>
   360→   360→            <ToggleGroupItem value="Incorporated City" className="h-8 px-3 text-[13px]">Cities</ToggleGroupItem>
   361→   361→            <ToggleGroupItem value="Unincorporated Urban Area" className="h-8 px-3 text-[13px]">
   362→   362→              Unincorp. UGAs
   363→   363→            </ToggleGroupItem>
   364→   364→          </ToggleGroup>
   365→   365→        </div>
   366→   366→
   367→   367→        <div className="ml-auto flex items-center gap-6 text-[13px] tabular-nums">
   368→   368→          {pins.length > 0 ? (
   369→   369→            <button
   370→   370→              type="button"
   371→   371→              onClick={() => setPins([])}
   372→   372→              title="Unpin all jurisdictions"
   373→   373→              className="inline-flex h-7 items-center gap-1.5 rounded-md border border-amber-400/60 bg-amber-400/10 px-2.5 text-[12px] font-medium text-amber-700 transition-all hover:bg-amber-400/20 active:scale-[0.97] dark:text-amber-300"
   374→   374→            >
   375→   375→              <Pin className="h-3 w-3" fill="currentColor" aria-hidden />
   376→   376→              {pins.length}/{MAX_PINS} pinned
   377→   377→              <PinOff className="ml-1 h-3 w-3" aria-hidden />
   378→   378→            </button>
   379→   379→          ) : null}
   380→   380→          <span className="text-muted-foreground">
   381→   381→            Showing <span className="font-semibold text-foreground">{rows.length}</span> of{" "}
   382→   382→            {submarkets.length}
   383→   383→          </span>
   384→   384→          <span className="hidden text-muted-foreground sm:block">
   385→   385→            Reserve: <span className="font-semibold text-foreground">{fmtAcres(Math.round(filteredNet))}</span>
   386→   386→          </span>
   387→   387→          <span className="hidden text-muted-foreground md:block">
   388→   388→            Avg CAGR: <span className="font-semibold text-emerald-600 dark:text-emerald-400">{fmtPct(filteredCagr)}</span>
   389→   389→          </span>
   390→   390→        </div>
   391→   391→      </div>
   392→   392→
   393→   393→      {/* Matrix table — min-width tracks visible columns so hiding
   394→   394→          columns actually tightens the layout instead of leaving
   395→   395→          a fixed-width scroll region. */}
   396→   396→      <div className="overflow-x-auto rounded-xl border bg-card shadow-sm">
   397→   397→        <Table
   398→   398→          className="min-w-[640px]"
   399→   399→          style={{ minWidth: 320 + visibleCount * 95 }}
   400→   400→        >
   401→   401→          <TableHeader>
   402→   402→            <TableRow className="bg-muted/50 hover:bg-muted/50">
   403→   403→              <TableHead className="w-[210px]">Market</TableHead>
   404→   404→              {visibleCols.jurisdiction ? (
   405→   405→                <TableHead>Jurisdiction &amp; Framework</TableHead>
   406→   406→              ) : null}
   407→   407→              {SORTABLE.map((c) => {
   408→   408→                const colKey: ColumnKey | null =
   409→   409→                  c.col === "net"
   410→   410→                    ? "net"
   411→   411→                    : c.col === "baseline"
   412→   412→                      ? "baseline"
   413→   413→                      : c.col === "dom"
   414→   414→                        ? "dom"
   415→   415→                        : c.col === "cagr"
   416→   416→                          ? "cagr"
   417→   417→                          : c.col === "forecast"
   418→   418→                            ? "forecast"
   419→   419→                            : "footprint";
   420→   420→                if (!visibleCols[colKey]) return null;
   421→   421→                return (
   422→   422→                  <TableHead key={c.col} className="text-right">
   423→   423→                    <button
   424→   424→                      type="button"
   425→   425→                      onClick={() => toggleSort(c.col)}
   426→   426→                      className={cn(
   427→   427→                        "inline-flex items-center gap-1 text-[11px] font-semibold uppercase tracking-[0.1em] transition-colors",
   428→   428→                        sortCol === c.col
   429→   429→                          ? "text-emerald-600 dark:text-emerald-400"
   430→   430→                          : "hover:text-foreground"
   431→   431→                      )}
   432→   432→                      aria-label={`Sort by ${c.label}`}
   433→   433→                    >
   434→   434→                      {c.label}
   435→   435→                      {sortCol === c.col ? (
   436→   436→                        sortDir === "asc" ? (
   437→   437→                          <ArrowUp className="h-3 w-3" aria-hidden />
   438→   438→                        ) : (
   439→   439→                          <ArrowDown className="h-3 w-3" aria-hidden />
   440→   440→                        )
   441→   441→                      ) : (
   442→   442→                        <ArrowUpDown className="h-3 w-3 opacity-40" aria-hidden />
   443→   443→                      )}
   444→   444→                    </button>
   445→   445→                  </TableHead>
   446→   446→                );
   447→   447→              })}
   448→   448→              {visibleCols.sqft ? (
   449→   449→                <TableHead className="text-right text-[11px] font-semibold uppercase tracking-[0.1em]">
   450→   450→                  $ / SqFt
   451→   451→                </TableHead>
   452→   452→              ) : null}
   453→   453→              {visibleCols.depletion ? (
   454→   454→                <TableHead className="text-right text-[11px] font-semibold uppercase tracking-[0.1em]">
   455→   455→                  Depletion
   456→   456→                </TableHead>
   457→   457→              ) : null}
   458→   458→              <TableHead className="w-10" aria-label="Row actions" />
   459→   459→            </TableRow>
