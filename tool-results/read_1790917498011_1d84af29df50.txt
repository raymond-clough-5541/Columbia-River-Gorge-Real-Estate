    98→const DEFAULT_VISIBLE: Record<ColumnKey, boolean> = {

   100→  footprint: true,

   102→  baseline: true,

   109→
   110→/** Row-pinning cap — three jurisdictions above the fold. */
   111→const MAX_PINS = 3;
   112→
   113→/** Session-scoped pin persistence — survives the "Model the pinned set →
   114→ *  ← Back to master matrix" round trip (view unmounts mid-loop). MatrixView
   115→ *  only mounts post-hydration (the hash router serves the overview server
   116→ *  snapshot first), so reading sessionStorage in the initializer is safe. */
   117→const PINS_KEY = "crgnsa-matrix-pins";
   118→
   119→function readPins(): string[] {
   120→  try {
   121→    const raw = sessionStorage.getItem(PINS_KEY);
   122→    const parsed = raw ? (JSON.parse(raw) as unknown) : [];
   123→    return Array.isArray(parsed)
   124→      ? parsed
   125→          .filter((x): x is string => typeof x === "string")
   126→          .slice(0, MAX_PINS)
   127→      : [];
   128→  } catch {
   129→    return [];
   130→  }
   131→}
   132→
   133→/** Text-only tier colors for the pinned-set CAGR figure — the same text
   134→ *  pairs the CAGR_TIER_STYLES badges use, minus their bg/border chrome
   135→ *  (the amber dossier strip supplies its own tint). */
   136→const CAGR_VALUE_COLORS: Record<CagrTier, string> = {
   137→  elite: "text-emerald-700 dark:text-emerald-300",
   138→  strong: "text-emerald-600 dark:text-emerald-400",
   139→  moderate: "text-amber-700 dark:text-amber-300",
   140→  baseline: "text-zinc-600 dark:text-zinc-300",
   141→};
   142→
   143→export function MatrixView({
   144→  submarkets,
   145→  navigate,
   146→}: {
   147→  submarkets: Submarket[];
   148→  navigate: NavigateFn;
   149→}) {
   150→  const [stateFilter, setStateFilter] = useState<"all" | "OR" | "WA">("all");
   151→  const [jurisdictionFilter, setJurisdictionFilter] = useState<
   152→    "all" | "Incorporated City" | "Unincorporated Urban Area"
   153→  >("all");
   154→  const [sortCol, setSortCol] = useState<SortCol>("net");
   155→  const [sortDir, setSortDir] = useState<SortDir>("desc");
   156→  const [detailSlug, setDetailSlug] = useState<string | null>(null);
   157→  const [visibleCols, setVisibleCols] =
   158→    useState<Record<ColumnKey, boolean>>(DEFAULT_VISIBLE);
   159→  const [pins, setPins] = useState<string[]>(readPins);
   160→
   161→  // Mirror pins into sessionStorage so the "Model the pinned set → back"
   162→  // round trip keeps the shortlist alive for the whole session.
   163→  useEffect(() => {
   164→    try {
   165→      if (pins.length === 0) sessionStorage.removeItem(PINS_KEY);
   166→      else sessionStorage.setItem(PINS_KEY, JSON.stringify(pins));
   167→    } catch {
   168→      /* storage unavailable — pins stay view-local */
   169→    }
   170→  }, [pins]);
   171→
   172→  const togglePin = (slug: string) => {
   173→    // Event-time logic only — never inside a state updater (that would
   174→    // update the Toaster while MatrixView renders → React warning).
   175→    if (pins.includes(slug)) {
   176→      setPins((prev) => prev.filter((s) => s !== slug));
   177→      return;
   178→    }
   179→    if (pins.length >= MAX_PINS) {
   180→      toast({
   181→        title: "Pin limit reached",
   182→        description: `Up to ${MAX_PINS} jurisdictions can stay pinned — unpin one first.`,
   183→      });
   184→      return;
   185→    }
   186→    setPins((prev) => (prev.includes(slug) ? prev : [...prev, slug]));
   187→  };
   188→
   189→  const visibleCount = useMemo(
   190→    () => ALL_COLUMNS.filter((c) => visibleCols[c]).length,
   191→    [visibleCols]
   192→  );
   193→  const toggleCol = (col: ColumnKey) =>
   194→    setVisibleCols((v) => ({ ...v, [col]: !v[col] }));
   195→
   196→  const rows = useMemo(() => {
   197→    const mid = (s: Submarket) =>
   198→      (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2;
   199→    const forecastMid = (s: Submarket) =>
   200→      (s.projectedPrice2046Min + s.projectedPrice2046Max) / 2;
   201→
   202→    const value = (s: Submarket): number => {
   203→      switch (sortCol) {
   204→        case "net":
   205→          return mid(s);
   206→        case "baseline":
   207→          return s.baselinePrice2026;
   208→        case "cagr":
   209→          return s.projectedCagr;
   210→        case "forecast":
   211→          return forecastMid(s);
   212→        case "footprint":
   213→          return s.totalFootprintAcres;
   214→        case "dom":
   215→          return (s.daysOnMarketMin + s.daysOnMarketMax) / 2;
   216→      }
   217→    };
   218→
   219→    return submarkets
   220→      .filter(
   221→        (s) =>
   222→          (stateFilter === "all" || s.state === stateFilter) &&
   223→          (jurisdictionFilter === "all" || s.jurisdictionType === jurisdictionFilter)
   224→      )
   225→      .sort((a, b) =>
   226→        sortDir === "asc" ? value(a) - value(b) : value(b) - value(a)
   227→      );
   228→  }, [submarkets, stateFilter, jurisdictionFilter, sortCol, sortDir]);
   229→
   230→  const filteredNet = rows.reduce(
   231→    (a, s) => a + (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2,
   232→    0
   233→  );
   234→  const filteredCagr =
   235→    rows.length > 0
   236→      ? rows.reduce((a, s) => a + s.projectedCagr, 0) / rows.length
   237→      : 0;
   238→
   239→  const detail = submarkets.find((s) => s.slug === detailSlug) ?? null;
   240→
   241→  /** Pinned jurisdictions float above the fold, in pin order. */
   242→  const orderedRows = useMemo(() => {
   243→    const pinnedRows = pins
   244→      .map((slug) => rows.find((r) => r.slug === slug))
   245→      .filter((r): r is Submarket => Boolean(r));
   246→    const rest = rows.filter((r) => !pins.includes(r.slug));
   247→    return [...pinnedRows, ...rest];
   248→  }, [rows, pins]);
   249→
   250→  /** Pinned-set aggregate dossier — every figure for the amber strip in a
   251→   *  single memo keyed on pins × submarkets. Null when nothing is pinned,
   252→   *  so the strip mounts/unmounts with pin state. */
   253→  const pinnedSet = useMemo(() => {
   254→    const pinned = pins
   255→      .map((slug) => submarkets.find((s) => s.slug === slug))
   256→      .filter((s): s is Submarket => Boolean(s));
   257→    if (pinned.length === 0) return null;
   258→    const mid = (s: Submarket) =>
   259→      (s.netBuildableAcresMin + s.netBuildableAcresMax) / 2;
   260→    const corridorReserve = submarkets.reduce((a, s) => a + mid(s), 0);
   261→    const combinedReserve = pinned.reduce((a, s) => a + mid(s), 0);
   262→    const avgCagr =
   263→      pinned.reduce((a, s) => a + s.projectedCagr, 0) / pinned.length;
   264→    // Strict < keeps the first-pinned market on depletion-year ties.
   265→    const earliest = pinned.reduce((a, s) =>
   266→      s.depletionYear < a.depletionYear ? s : a
   267→    );
   268→    return {
   269→      names: pinned.map((s) => s.name).join(" · "),
   270→      combinedReserve,
   271→      corridorShare:
   272→        corridorReserve > 0 ? (combinedReserve / corridorReserve) * 100 : 0,
   273→      avgCagr,
   274→      cagrMin: Math.min(...pinned.map((s) => s.projectedCagr)),
   275→      cagrMax: Math.max(...pinned.map((s) => s.projectedCagr)),
   276→      blendedMultiple: Math.pow(1 + avgCagr / 100, 20),
   277→      earliestYear: earliest.depletionYear,
   278→      earliestName: earliest.name,
   279→      runway: earliest.depletionYear - 2026,
   280→      critical: earliest.depletionYear <= 2032,
   281→    };
   282→  }, [pins, submarkets]);
   283→
   284→  const { toast } = useToast();
   285→
   286→  const exportCsv = () => {
   287→    const headers = [
   288→      "Market",
   289→      "State",
   290→      "County",
   291→      "Jurisdiction Type",
   292→      "Regulatory Framework",
   293→      "UGB/UGA Footprint (ac)",
   294→      "Gross Vacant (ac)",
   295→      "Net Buildable Min (ac)",
   296→      "Net Buildable Max (ac)",
   297→      "Net Buildable Mid (ac)",
   298→      "2026 Baseline Median ($)",
   299→      "Price/SqFt Min ($)",
   300→      "Price/SqFt Max ($)",
   301→      "Days on Market Min",
   302→      "Days on Market Max",
   303→      "Projected 20-yr CAGR (%)",
   304→      "2046 Forecast Min ($)",
   305→      "2046 Forecast Max ($)",
   306→      "Est. Raw-Land Depletion Year",
   307→      "Water Purveyor",
   308→      "Wastewater System",
   309→      "Primary Constraints & WUI Risk",
   310→    ];
   311→    const data: CsvCell[][] = orderedRows.map((s) => [
   312→      s.name,
   313→      s.state,
   314→      s.county,
   315→      s.jurisdictionType,
   316→      s.regulatoryFramework,
   317→      s.totalFootprintAcres,
   318→      s.grossVacantAcres,
   319→      s.netBuildableAcresMin,
   320→      s.netBuildableAcresMax,
   321→      Math.round(((s.netBuildableAcresMin + s.netBuildableAcresMax) / 2) * 10) / 10,
   322→      s.baselinePrice2026,
   323→      Math.round(s.pricePerSqftMin),
   324→      Math.round(s.pricePerSqftMax),
   325→      s.daysOnMarketMin,
   326→      s.daysOnMarketMax,
   327→      s.projectedCagr,
   328→      s.projectedPrice2046Min,
   329→      s.projectedPrice2046Max,
   330→      s.depletionYear,
   331→      s.waterPurveyor,
   332→      s.wastewaterSystem,
   333→      s.primaryConstraints,
   334→    ]);
   335→    downloadCsv(
   336→      `crgnsa-master-matrix-${timestampSuffix()}.csv`,
   337→      toCsv(headers, data)
   338→    );
   339→    toast({
   340→      title: "Master Matrix exported",
   341→      description: `${rows.length} jurisdiction${rows.length === 1 ? "" : "s"} · current filters and sort order applied.`,
   342→    });
   343→  };
   344→
   345→  const toggleSort = (col: SortCol) => {
   346→    if (col === sortCol) {
   347→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   348→    } else {
   349→      setSortCol(col);
   350→      setSortDir("desc");
   351→    }
   352→  };
   353→
   354→  return (
   355→    <div className="mx-auto w-full max-w-7xl px-4 pb-16 pt-10 sm:px-6 lg:px-8">
   356→      <SectionHeader
   357→        eyebrow="Master Real Estate & Land Inventory Matrix"
   358→        title="All 11 jurisdictions, one sortable ledger"
   359→        description="Filter by state and jurisdiction type, sort any column, and click a row for the infrastructure dossier — water purveyors, wastewater capacity, and fire-risk constraints that gate every buildable acre."
   360→        action={
   361→          <div className="flex items-center gap-2">
   362→            <DropdownMenu>
   363→              <DropdownMenuTrigger asChild>
   364→                <button
   365→                  type="button"
   366→                  aria-label="Toggle column visibility"
   367→                  className="inline-flex h-9 items-center gap-2 rounded-lg border bg-card px-3.5 text-[13px] font-medium shadow-sm transition-all hover:border-zinc-400 hover:shadow dark:hover:border-zinc-600 active:scale-[0.98]"
   368→                >
   369→                  <Columns3 className="h-3.5 w-3.5" aria-hidden />
   370→                  Columns
   371→                  <span className="rounded-sm bg-muted px-1.5 text-[11px] font-semibold tabular-nums">
   372→                    {visibleCount}/{ALL_COLUMNS.length}
   373→                  </span>
   374→                </button>
   375→              </DropdownMenuTrigger>
   376→              <DropdownMenuContent align="end" className="w-56">
   377→                <DropdownMenuLabel className="text-[12px] uppercase tracking-wider">
   378→                  Visible columns
   379→                </DropdownMenuLabel>
   380→                <DropdownMenuSeparator />
   381→                {ALL_COLUMNS.map((col) => (
   382→                  <DropdownMenuCheckboxItem
   383→                    key={col}
   384→                    checked={visibleCols[col]}
   385→                    onCheckedChange={() => toggleCol(col)}
   386→                    onSelect={(e) => e.preventDefault()}
   387→                    className="text-[13px]"
   388→                  >
   389→                    {COLUMN_LABELS[col]}
   390→                  </DropdownMenuCheckboxItem>
   391→                ))}
   392→              </DropdownMenuContent>
   393→            </DropdownMenu>
   394→            <button
   395→              type="button"
   396→              onClick={exportCsv}
   397→              disabled={rows.length === 0}
   398→              className="inline-flex h-9 items-center gap-2 rounded-lg border bg-card px-4 text-[13px] font-medium shadow-sm transition-all hover:border-zinc-400 hover:shadow disabled:cursor-not-allowed disabled:opacity-50 dark:hover:border-zinc-600 active:scale-[0.98]"
   399→            >
   400→              <Download className="h-3.5 w-3.5" aria-hidden />
   401→              Export CSV
   402→            </button>
   403→          </div>
   404→        }
   405→      />
   406→
   407→      {/* Filter + summary bar */}
   408→      <div className="mb-5 flex flex-wrap items-center gap-4 rounded-xl border bg-card p-4 shadow-sm">
   409→        <div className="flex items-center gap-3">
   410→          <MicroLabel className="hidden sm:block">State</MicroLabel>
   411→          <ToggleGroup
   412→            type="single"
   413→            value={stateFilter}
   414→            onValueChange={(v) => {
   415→              if (v) setStateFilter(v as "all" | "OR" | "WA");
   416→            }}
   417→            className="gap-1"
   418→          >
   419→            <ToggleGroupItem value="all" className="h-8 px-3 text-[13px]">All</ToggleGroupItem>
   420→            <ToggleGroupItem value="OR" className="h-8 px-3 text-[13px]">Oregon</ToggleGroupItem>
   421→            <ToggleGroupItem value="WA" className="h-8 px-3 text-[13px]">Washington</ToggleGroupItem>
   422→          </ToggleGroup>
   423→        </div>
   424→        <div className="flex items-center gap-3">
   425→          <MicroLabel className="hidden sm:block">Jurisdiction</MicroLabel>
   426→          <ToggleGroup
   427→            type="single"
   428→            value={jurisdictionFilter}
   429→            onValueChange={(v) => {
   430→              if (v)
   431→                setJurisdictionFilter(
   432→                  v as "all" | "Incorporated City" | "Unincorporated Urban Area"
   433→                );
   434→            }}
   435→            className="gap-1"
   436→          >
   437→            <ToggleGroupItem value="all" className="h-8 px-3 text-[13px]">All</ToggleGroupItem>
   438→            <ToggleGroupItem value="Incorporated City" className="h-8 px-3 text-[13px]">Cities</ToggleGroupItem>
   439→            <ToggleGroupItem value="Unincorporated Urban Area" className="h-8 px-3 text-[13px]">
   440→              Unincorp. UGAs
   441→            </ToggleGroupItem>
   442→          </ToggleGroup>
   443→        </div>
   444→
   445→        <div className="ml-auto flex items-center gap-6 text-[13px] tabular-nums">
   446→          {pins.length > 0 ? (
   447→            <button
   448→              type="button"
   449→              onClick={() => setPins([])}
   450→              title="Unpin all jurisdictions"
   451→              className="inline-flex h-7 items-center gap-1.5 rounded-md border border-amber-400/60 bg-amber-400/10 px-2.5 text-[12px] font-medium text-amber-700 transition-all hover:bg-amber-400/20 active:scale-[0.97] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/60 dark:text-amber-300"
   452→            >
   453→              <Pin className="h-3 w-3" fill="currentColor" aria-hidden />
   454→              {pins.length}/{MAX_PINS} pinned
   455→              <PinOff className="ml-1 h-3 w-3" aria-hidden />
   456→            </button>
   457→          ) : null}
   458→          <span className="text-muted-foreground">
   459→            Showing <span className="font-semibold text-foreground">{rows.length}</span> of{" "}
   460→            {submarkets.length}
   461→          </span>
   462→          <span className="hidden text-muted-foreground sm:block">
   463→            Reserve: <span className="font-semibold text-foreground">{fmtAcres(Math.round(filteredNet))}</span>
   464→          </span>
   465→          <span className="hidden text-muted-foreground md:block">
   466→            Avg CAGR: <span className="font-semibold text-emerald-600 dark:text-emerald-400">{fmtPct(filteredCagr)}</span>
   467→          </span>
   468→        </div>
   469→      </div>
   470→
   471→      {/* Pinned-set aggregate dossier — mounts only while ≥1 pin exists
   472→          and unmounts with the last one; all figures derive from the
   473→          pinnedSet memo above. */}
   474→      {pinnedSet ? (
   475→        <section
   476→          role="region"
   477→          aria-label="Pinned set aggregate"
   478→          className="mb-5 rounded-xl border border-amber-400/50 bg-amber-400/[0.06] p-3.5 shadow-sm sm:p-4"
   479→        >
   480→          <div className="flex flex-wrap items-center gap-x-6 gap-y-3.5">
   481→            {/* Left block — pinned markets in pin order */}
   482→            <div className="min-w-0 flex-auto">
   483→              <p className="flex items-center gap-1.5 text-[10.5px] font-semibold uppercase tracking-wider text-amber-700 dark:text-amber-300">
   484→                <Pin
   485→                  className="h-3 w-3"
   486→                  fill="currentColor"
   487→                  aria-hidden
   488→                />
   489→                Pinned set
   490→              </p>
   491→              <p className="mt-1 min-w-0 flex-wrap text-[13px] font-semibold leading-snug">
   492→                {pinnedSet.names}
   493→              </p>
   494→            </div>
   495→
   496→            {/* Right block — compact aggregate tiles; wrap into a 2×2
   497→                grid on narrow viewports via flex-wrap + min-w. */}
   498→            <div className="flex flex-wrap gap-x-5 gap-y-3">
   499→              <div className="min-w-[130px]">
   500→                <p className="text-[10.5px] font-semibold uppercase tracking-wider text-muted-foreground">
   501→                  Combined reserve
   502→                </p>
   503→                <p className="mt-0.5 text-[15px] font-semibold leading-tight tabular-nums">
   504→                  {fmtAcres(Math.round(pinnedSet.combinedReserve))}
   505→                </p>
   506→                <p className="mt-0.5 text-[11px] leading-snug text-muted-foreground tabular-nums">
   507→                  {fmtPct(pinnedSet.corridorShare)} of corridor reserve
   508→                </p>
   509→              </div>
   510→
   511→              <div className="min-w-[130px] border-l border-amber-400/30 pl-4">
   512→                <p className="text-[10.5px] font-semibold uppercase tracking-wider text-muted-foreground">
   513→                  Avg 20-yr CAGR
   514→                </p>
   515→                <p
   516→                  className={cn(
   517→                    "mt-0.5 text-[15px] font-semibold leading-tight tabular-nums",
   518→                    CAGR_VALUE_COLORS[cagrTier(pinnedSet.avgCagr)]
   519→                  )}
   520→                >
   521→                  {fmtPct(pinnedSet.avgCagr)}
   522→                </p>
   523→                <p className="mt-0.5 text-[11px] leading-snug text-muted-foreground tabular-nums">
   524→                  band {pinnedSet.cagrMin.toFixed(1)}–{pinnedSet.cagrMax.toFixed(1)}%
   525→                </p>
