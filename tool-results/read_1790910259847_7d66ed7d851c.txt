    57→    57→interface SavedScenario {
    58→    58→  id: string;
    59→    59→  name: string;

    68→    68→const STORAGE_KEY = "crgnsa-saved-scenarios";
    69→    69→const CHANGE_EVENT = "crgnsa:saved-scenarios-changed";
    70→    70→const EMPTY: SavedScenario[] = [];
    71→    71→
    72→    72→let cachedRaw: string | null = null;
    73→    73→let cachedList: SavedScenario[] = EMPTY;
    74→    74→
    75→    75→function readSaved(): SavedScenario[] {
    76→    76→  let raw: string | null = null;
    77→    77→  try {
    78→    78→    raw = window.localStorage.getItem(STORAGE_KEY);
    79→    79→  } catch {
    80→    80→    raw = null;
    81→    81→  }
    82→    82→  if (raw === cachedRaw) return cachedList;
    83→    83→  cachedRaw = raw;
    84→    84→  try {
    85→    85→    const parsed = raw ? (JSON.parse(raw) as SavedScenario[]) : [];
    86→    86→    cachedList = Array.isArray(parsed) ? parsed : EMPTY;
    87→    87→  } catch {
    88→    88→    cachedList = EMPTY;
    89→    89→  }
    90→    90→  return cachedList;
    91→    91→}
    92→    92→
    93→    93→function writeSaved(list: SavedScenario[]): void {
    94→    94→  try {
    95→    95→    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
    96→    96→    window.dispatchEvent(new Event(CHANGE_EVENT));
    97→    97→  } catch {
    98→    98→    /* storage unavailable — ignore */
    99→    99→  }
   100→   100→}
   101→   101→
   102→   102→function useSavedScenarios(): SavedScenario[] {
   103→   103→  return useSyncExternalStore(
   104→   104→    (cb) => {
   105→   105→      window.addEventListener(CHANGE_EVENT, cb);
   106→   106→      window.addEventListener("storage", cb);
   107→   107→      return () => {
   108→   108→        window.removeEventListener(CHANGE_EVENT, cb);
   109→   109→        window.removeEventListener("storage", cb);
   110→   110→      };
   111→   111→    },
   112→   112→    readSaved,
   113→   113→    () => EMPTY
   114→   114→  );
   115→   115→}

   124→   124→function ChartTooltip({
   125→   125→  active,
   126→   126→  payload,
   127→   127→  label,
   128→   128→  depletionBySlug,
   129→   129→}: {
   130→   130→  active?: boolean;
   131→   131→  payload?: TooltipEntry[];
   132→   132→  label?: string | number;
   133→   133→  depletionBySlug: Record<string, number>;
   134→   134→}) {
   135→   135→  if (!active || !payload || payload.length === 0) return null;
   136→   136→  const year = Number(label);
   137→   137→  return (
   138→   138→    <div className="rounded-lg border bg-popover p-3.5 text-popover-foreground shadow-xl">
   139→   139→      <p className="text-[12px] font-semibold uppercase tracking-wider text-muted-foreground">
   140→   140→        {year}
   141→   141→      </p>
   142→   142→      <div className="mt-2 space-y-1.5">
   143→   143→        {payload.map((entry) => {
   144→   144→          const value = entry.value ?? 0;
   145→   145→          const isScenario = entry.dataKey === "scenario";
   146→   146→          const depletion = isScenario
   147→   147→            ? null
   148→   148→            : depletionBySlug[String(entry.dataKey)];
   149→   149→          const depleted = depletion !== null && depletion !== undefined && year >= depletion;
   150→   150→          return (
   151→   151→            <div key={String(entry.dataKey)} className="flex items-center gap-2.5 text-[13px]">
   152→   152→              <span
   153→   153→                className="h-2.5 w-2.5 shrink-0 rounded-full"
   154→   154→                style={{ backgroundColor: entry.color }}
   155→   155→                aria-hidden
   156→   156→              />
   157→   157→              <span className="font-medium">{entry.name}</span>
   158→   158→              <span className="ml-auto pl-4 font-semibold tabular-nums">
   159→   159→                {fmtCurrency(value, { compact: true })}
   160→   160→              </span>
   161→   161→              {depleted ? (
   162→   162→                <Flame className="h-3.5 w-3.5 shrink-0 text-rose-500" aria-label="Raw land depleted" />
   163→   163→              ) : null}
   164→   164→            </div>
   165→   165→          );
   166→   166→        })}
   167→   167→      </div>
   168→   168→      <p className="mt-2 border-t pt-1.5 text-[11px] text-muted-foreground">
   169→   169→        🜂 marks years after a market's projected raw-land exhaustion
   170→   170→      </p>
   171→   171→    </div>
   172→   172→  );
   173→   173→}

   175→   175→export function ProjectionsView({ submarkets }: { submarkets: Submarket[] }) {

   198→   198→        row[s.slug] = futureValue(s.baselinePrice2026, s.projectedCagr, n);
   199→   199→      }

   201→   201→        row.scenario = futureValue(pv, cagr, n);
   202→   202→      }

   305→   305→    });
   306→   306→  };
